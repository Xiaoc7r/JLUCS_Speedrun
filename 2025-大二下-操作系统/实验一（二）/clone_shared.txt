#define _GNU_SOURCE
#include <sched.h>
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <semaphore.h>
#include <unistd.h>
#include <sys/wait.h>

int producer(void *args);
int consumer(void *args);
pthread_mutex_t mutex;  // 互斥锁
sem_t product;          // 产品信号量
sem_t warehouse;        // 仓库空间信号量
char buffer[8][4];      // 共享缓冲区
int bp = 0;             // 缓冲区指针

int main(int argc, char **argv)
{
    pthread_mutex_init(&mutex, NULL);
    sem_init(&product, 0, 0);
    sem_init(&warehouse, 0, 8);
    int clone_flag, retval;
    char *stack;
    clone_flag = CLONE_VM | CLONE_SIGHAND | CLONE_FS | CLONE_FILES;
    int i;
    for(i = 0; i < 2; i++) /*创建4个线程*/
    {
        int *arg = malloc(sizeof(int));
        *arg = i;
        stack = (char*)malloc(4096);
        retval = clone((void*)producer, &(stack[4095]), clone_flag, (void*)arg);
        if(retval == -1) {
            perror("clone producer failed");
            exit(1);
        }
        
        int *arg2 = malloc(sizeof(int));
        *arg2 = i;
        stack = (char*)malloc(4096);
        sleep(1);
        retval = clone((void*)consumer, &(stack[4095]), clone_flag, (void*)arg2);
        if(retval == -1) {
            perror("clone consumer failed");
            exit(1);
        }
    }
    
    
    
    sleep(100);
    
   
    exit(0);
}

// 生产者线程函数
int producer(void *args)
{
    int id = *((int*)args);
    
    for(int i = 0; i < 10; i++)
    {
        sleep(i+1);  // 线程速度差异
        
        sem_wait(&warehouse);       // P操作：等待仓库有空位
        pthread_mutex_lock(&mutex); // 加锁，确保互斥访问缓冲区
        
        // 生产数据放入共享缓冲区
        if(id == 0)
            strcpy(buffer[bp], "aaa\0");
        else
            strcpy(buffer[bp], "bbb\0");
        
        bp++;  // 更新缓冲区位置
        printf("producer%d produce %s in %d \n", id, buffer[bp-1], bp-1);
        
        pthread_mutex_unlock(&mutex); // 解锁
        sem_post(&product);           // V操作：通知有新产品
    }
    
    printf("producer%d is over!\n", id);
    return 0;
}

// 消费者线程函数
int consumer(void *args)
{
    int id = *((int*)args);
    
    for(int i = 0; i < 10; i++)
    {
        sleep(10-i);  // 线程速度差异
        
        sem_wait(&product);         // P操作：等待有产品
        pthread_mutex_lock(&mutex); // 加锁，确保互斥访问缓冲区
        
        // 从共享缓冲区取出数据
        bp--;
        printf("consumer%d get %s in %d\n", id, buffer[bp], bp);
        strcpy(buffer[bp], "zzz\0"); // 标记已取出
        
        pthread_mutex_unlock(&mutex); // 解锁
        sem_post(&warehouse);         // V操作：通知有空位
    }
    
    printf("consumer%d is over!\n", id);
    return 0;
}
