#include "math.h"
#include "sched.h"
#include "pthread.h"
#include "stdlib.h"
#include "semaphore.h"
#include "stdio.h"

typedef struct{  /* 实时任务描述 */
    char task_id;
    int call_num;  /* 任务发生次数 */
    int ci;  /* Ci */
    int ti;  /* Ti */
    int ci_left;
    int ti_left;
    int flag;  /* 任务是否活跃,0 为否,2 为是 */
    int arg;  /* 参数 */
    pthread_t th;  /* 任务对应线程 */
} task;

void proc(int * args);
void * idle();
int select_proc();

int task_num = 0;
int idle_num = 0;

int alg;  /* 所选算法,1 为 EDF,2 为 RMS,                      RR+++3 */
int curr_proc = -1;
int demo_time = 100;  /* 演示时间 */

task * tasks;
pthread_mutex_t proc_wait[100];
pthread_mutex_t main_wait,idle_wait;
float sum = 0;
pthread_t idle_proc;
int rr_last = -1;                                   //轮询调度，上次选择的任务索引 

int main(int argc,char** argv)
{
    pthread_mutex_init(&main_wait,NULL);
    pthread_mutex_lock(&main_wait);  /* 下次执行 lock 等待 */
    pthread_mutex_init(&idle_wait,NULL);
    pthread_mutex_lock(&idle_wait);  /* 下次执行 lock 等待 */
    printf("Please input number of real time tasks:\n");
    scanf("%d",&task_num);
    tasks = (task *)malloc(task_num * sizeof(task));
    int i;
    for(i = 0;i < task_num;i ++)
    {
        pthread_mutex_init(&proc_wait[i],NULL);
        pthread_mutex_lock(&proc_wait[i]);
    }
    for(i = 0;i < task_num;i ++)
    {
        printf("Please input task id,followed by Ci and Ti:\n");
        getchar();
        scanf("%c,%d,%d,",&tasks[i].task_id,&tasks[i].ci,&tasks[i].ti);
        tasks[i].ci_left = tasks[i].ci;
        tasks[i].ti_left = tasks[i].ti;
        tasks[i].flag = 2;
        tasks[i].arg = i;
        tasks[i].call_num = 1;
        sum = sum + (float)tasks[i].ci/(float)tasks[i].ti;
    }
    printf("Please input algorithm,1 for EDF,2 for RMS,3 for RR:");
    getchar();
    scanf("%d",&alg);
    printf("Please input demo time:");
    scanf("%d",&demo_time);
    double r = 1;  /* EDF 算法 */
    if(alg == 2)
    { /* RMS 算法 */
        r = ((double)task_num) * (exp(log(2)/(double)task_num) - 1);
        printf("r is %lf\n",r);
    }
    if(sum > r)
    { /* 不可调度 */
        printf("(sum = %lf>r = %lf),not schedulable! \n",sum,r);
        exit(2);
    }
    pthread_create(&idle_proc,NULL,(void *)idle,NULL);/* 创建闲逛线程 */
    for(i = 0;i < task_num;i ++)  /* 创建实时任务线程 */
        pthread_create(&tasks[i].th,NULL,(void *)proc,&tasks[i].arg);
    for(i = 0;i < demo_time;i ++)
    {
        int j;
        if((curr_proc = select_proc(alg))!= -1)
        { /* 按调度算法选线程 */
            pthread_mutex_unlock(&proc_wait[curr_proc]);  /* 唤醒 */
            pthread_mutex_lock(&main_wait);  /* 主线程等待 */
        }
        else
        { /* 无可运行任务,选择闲逛线程 */
            pthread_mutex_unlock(&idle_wait);
            pthread_mutex_lock(&main_wait);
        }
        for(j = 0;j < task_num;j ++)
        {
            /* Ti --,为 0 时开始下一周期 */
            if(--tasks[j].ti_left == 0)
            {
                tasks[j].ti_left = tasks[j].ti;
                tasks[j].ci_left = tasks[j].ci;
                pthread_create(&tasks[j].th,NULL,(void *)proc,&tasks[j].arg);
                tasks[j].flag = 2;
            }
        }
    }
    printf("\n");
    sleep(10);
};


void proc(int * args)
{
    while(tasks[ * args].ci_left > 0)  // 任务剩余执行时间大于0时循环
    {
        pthread_mutex_lock(&proc_wait[ * args]);  /* 等待主控线程调度：确保每个时间单位只有一个任务执行； */

        if(idle_num!= 0)
        {
            printf("idle(% d)",idle_num);  
            idle_num = 0;  
        }
        

        printf("%c%d",tasks[ * args].task_id,tasks[ * args].call_num);  //输出任务ID和调用次数
        tasks[ * args].ci_left --;  /* 执行一个时间单位：
                                      1. 剩余执行时间减1
                                      2. 对应输出中的一个字符（如a1, b2） */
        
        // 任务执行完毕处理
        if(tasks[ * args].ci_left == 0)
        {
            printf("(%d)",tasks[ * args].ci);  //（(10)）
            tasks[ * args].flag = 0;  //标记任务为"完成"状态
            tasks[ * args].call_num ++;  //调用次数递增，为下一周期做准备
        }
        
        pthread_mutex_unlock(&main_wait);/* 唤醒主线程：
                                             1. 通知主控线程当前时间单位已执行完毕
                                             2. 主控线程可继续调度下一个任务 */
    }
};


void * idle()
{
    while(1)  // 无限循环，持续检查是否需要进入空闲状态
    {
        pthread_mutex_lock(&idle_wait);  /* 等待被调度：
                                              1. 主线程通过解锁此锁来激活空闲状态
                                              2. 避免在有任务可执行时进入空闲 */
        
        printf("->");  
        idle_num ++;  // 空闲时间计数器递增
        pthread_mutex_unlock(&main_wait);  /* 唤醒主控线程：
                                               1. 通知主控线程空闲时间单位已执行完毕
                                               2. 主控线程可继续尝试调度任务 */
    }
};

int select_proc(int alg)
{
    int j;
    int temp1,temp2;
    temp1 = 10000;
    temp2 = -1;
    if((alg == 2)&&(curr_proc!= -1)&&(tasks[curr_proc].flag!= 0))
        return curr_proc;
    for(j = 0;j < task_num;j ++)
    {
        if(tasks[j].flag == 2)
        {
            switch(alg)
            {
                case 1:  /* EDF 算法 */
                    if(temp1 > tasks[j].ti_left)
                    {
                        temp1 = tasks[j].ti_left;
                        temp2 = j;
                    }
                    break;
                case 2:  /* RMS 算法 */
                    if(temp1 > tasks[j].ti)
                    {
                        temp1 = tasks[j].ti;
                        temp2 = j;
                    }
                    break;


                case 3:  //RR 算法，记录上次调度的任务索引
                    rr_last = (rr_last + 1) % task_num;
                    if(tasks[rr_last].flag == 2)
                    {
                        temp2 = rr_last;
                    }
                    else
                    {
                        int k;
                        for(k = 0; k < task_num; k++)
                        {
                            int idx = (rr_last + k) % task_num;
                            if(tasks[idx].flag == 2)
                            {
                                rr_last = idx;
                                temp2 = idx;
                                break;
                            }
                        }
                    }
                    break;



            }
        }
    }
    return temp2;
}
