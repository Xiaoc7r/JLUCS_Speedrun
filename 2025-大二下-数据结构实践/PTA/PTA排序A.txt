C++ STL是Standard Template Library的简称，即标准模板库。简单来说，STL将常用的数据结构与算法进行了封装，用户需要时可以直接调用，不用重新开发。排序算法sort( )是STL包含的一个重要算法。

STL中的sort()函数基于快速排序算法实现，众所众知，快速排序是目前已知平均情况下最快的排序算法，被IEEE评选为20世纪十大算法之一，但其最坏情况下时间复杂度会退化为O(n 
2
 )。STL中的sort()对传统快速排序做了巧妙的改进，使其最坏情况下时间复杂度也能维持在O(nlogn)，它是如何实现的呢？

（1）快速排序算法最坏情况下时间复杂度退化为O(n 
2
 )的主要原因是，每次划分（Partition）操作时，都分在子数组的最边上，导致递归深度恶化为O(n)层。而STL的sort()在Partition操作有恶化倾向时，能够自我侦测，转而改为堆排序，使效率维持在堆排序的O(nlogn)。其具体方法是：侦测快速排序的递归深度，当递归深度达到⌊2log 
2
​
 n⌋=O(logn)层时，强行停止递归，转而对当前处理的子数组进行堆排序。

（2）此外，传统的快速排序在数据量很小时，为极小的子数组产生许多的递归调用，得不偿失。为此，STL的sort()进行了优化，在小数据量的情况下改用插入排序。具体做法是：当递归处理的子数组长度（子数组包含的元素个数）小于等于某个阈值threshold 时，停止处理并退出本层递归，使当前子数组停留在“接近排序但尚未完成”的状态，最后待所有递归都退出后，再对整个序列进行一次插入排序（注意不是对当前处理的子数组进行插入排序，而是在快速排序的所有递归完全退出后，对整个数组统一进行一次插入排序）。实验表明，此种策略有着良好的效率，因为插入排序在面对“接近有序”的序列时拥有良好的性能。

备注：Partition操作选取第1个元素作为基准元素。Partition操作的不同实现可能导致不同的输出结果，为保证输出结果唯一，该操作的实现请以教材为准，即Hoare提出快速排序算法时最早给出的Partition实现。堆排序的实现以老师课堂讲的算法为准。

//朱老师PPT_Partition
int Partition(int R[], int m, int n) { //对子数组Rm…Rn分划
    int K = R[m], L = m + 1, G = n; //Rm为基准元素
    int temp; //用于交换元素

    while (L <= G) {
        while (L <= n && R[L] <= K) L++; //从左向右找第一个>K的元素
        while (R[G] > K) G--; //从右向左找第一个≤K的元素
        if (L < G) {
            temp = R[L];
            R[L] = R[G];
            R[G] = temp;
            L++;
            G--;
        }
    }
    temp = R[m];
    R[m] = R[G];
    R[G] = temp;
    return G;
}


void MaxHeapify(int R[], int i, int heap_size) {
    int largest = i;
    int left = 2 * i;
    int right = 2 * i + 1;
    int temp;

    if (left <= heap_size && R[left] > R[largest])
        largest = left;

    if (right <= heap_size && R[right] > R[largest])
        largest = right;

    if (largest != i) {
        //交换R[i]和R[largest]
        temp = R[i];
        R[i] = R[largest];
        R[largest] = temp;
        MaxHeapify(R, largest, heap_size);
    }
}

void BuildMaxHeap(int R[], int heap_size) {
    int i;
    for (i = heap_size / 2; i >= 1; i--) {
        MaxHeapify(R, i, heap_size);
    }
}
void HeapSort(int R[], int m, int n) {
    int length = n - m + 1;
    int* temp;
    int i;
    int temp_swap;
    temp = (int*)malloc((length + 1) * sizeof(int));
    for (i = 1; i <= length; i++) {
        temp[i] = R[m + i - 1];
    }

    //建立初始堆
    BuildMaxHeap(temp, length);
    printf("Heap:");
    for (i = 1; i <= length; i++) {
        printf("%d ", temp[i]);
    }
    printf("\n");


    for (i = length; i > 1; i--) {
        // 交换temp[1]和temp[i]
        temp_swap = temp[1];
        temp[1] = temp[i];
        temp[i] = temp_swap;
        MaxHeapify(temp, 1, i - 1);
    }

    for (i = 1; i <= length; i++) {
        R[m + i - 1] = temp[i];
    }
}

//插入排序
void InsertionSort(int R[], int n) { // 对R[1]...R[n]排序
    int i, j, K;
    for (i = 2; i <= n; i++) { // R[1]..... R[i-1]  R[i]
        K = R[i];
        j = i - 1;
        while (j >= 1 && R[j] > K) {
            R[j + 1] = R[j];
            j--;
        }
        R[j + 1] = K;
    }
}
//朱老师PPT_QuickSort
void QuickSort(int R[], int m, int n, int max_depth, int current_depth) {
    extern int threshold; //外变量
    int k;
    if (n - m + 1 <= threshold) {
        return;
    }

    //递归深度达到上限使用堆排序
    if (current_depth >= max_depth) {
        HeapSort(R, m, n);
        return;
    }

    if (m < n) {
        k = Partition(R, m, n);
        QuickSort(R, m, k - 1, max_depth, current_depth + 1);
        QuickSort(R, k + 1, n, max_depth, current_depth + 1);
    }
}

/* 请在这里补充你的代码，即你所实现的sort函数 */
//朱老师PPT_Partition
int Partition(int R[], int m, int n) { //对子数组Rm…Rn分划
    int K = R[m], L = m + 1, G = n; //选第一个元素作为基准元素
    int temp; //用于交换元素

    while (L <= G) {
        while (L <= n && R[L] <= K) L++; //从左边找比基准元素大的数
        while (R[G] > K) G--; //从右边找比基准元素小的或等于的数
        if (L < G) {
            // 交换找到的两个数，让小的在左边，大的在右边
            temp = R[L];
            R[L] = R[G];
            R[G] = temp;
            L++;
            G--;
        }
    }
    // 把基准元素放到中间位置，这样基准左边的都比它小，右边的都比它大
    temp = R[m];
    R[m] = R[G];
    R[G] = temp;
    return G; // 返回基准元素最后的位置
}

// 维护最大堆的性质，确保父节点大于子节点
void MaxHeapify(int R[], int i, int heap_size) {
    int largest = i; // 假设当前节点是最大的
    int left = 2 * i; // 左孩子位置
    int right = 2 * i + 1; // 右孩子位置
    int temp;

    // 如果左孩子存在且比当前节点大
    if (left <= heap_size && R[left] > R[largest])
        largest = left;

    // 如果右孩子存在且比目前找到的最大值还大
    if (right <= heap_size && R[right] > R[largest])
        largest = right;

    // 如果最大值不是当前节点，则交换位置并继续调整
    if (largest != i) {
        temp = R[i];
        R[i] = R[largest];
        R[largest] = temp;
        // 递归调整被交换的子树
        MaxHeapify(R, largest, heap_size);
    }
}

// 建立最大堆，让整个数组符合堆的性质
void BuildMaxHeap(int R[], int heap_size) {
    int i;
    // 从最后一个非叶子节点开始向上调整
    for (i = heap_size / 2; i >= 1; i--) {
        MaxHeapify(R, i, heap_size);
    }
}

// 用堆排序对子数组进行排序
void HeapSort(int R[], int m, int n) {
    int length = n - m + 1; // 子数组长度
    int* temp; // 临时数组
    int i;
    int temp_swap;
    // 分配临时数组空间
    temp = (int*)malloc((length + 1) * sizeof(int));
    // 复制子数组到临时数组
    for (i = 1; i <= length; i++) {
        temp[i] = R[m + i - 1];
    }

    // 建立初始堆并打印堆内容
    BuildMaxHeap(temp, length);
    printf("Heap:");
    for (i = 1; i <= length; i++) {
        printf("%d ", temp[i]);
    }
    printf("\n");

    // 堆排序：每次取出最大值放到末尾，然后调整剩余部分
    for (i = length; i > 1; i--) {
        // 交换堆顶（最大值）和当前堆的最后一个元素
        temp_swap = temp[1];
        temp[1] = temp[i];
        temp[i] = temp_swap;
        // 调整剩余堆，保持堆的性质
        MaxHeapify(temp, 1, i - 1);
    }

    // 将排好序的数据复制回原数组
    for (i = 1; i <= length; i++) {
        R[m + i - 1] = temp[i];
    }
}

// 插入排序算法 - 对较小的数组效率很高
void InsertionSort(int R[], int n) { // 对R[1]...R[n]排序
    int i, j, K;
    for (i = 2; i <= n; i++) { // 假设R[1]已经排好序，从R[2]开始插入
        K = R[i]; // 当前要插入的元素
        j = i - 1;
        // 寻找插入位置并移动元素
        while (j >= 1 && R[j] > K) {
            R[j + 1] = R[j]; // 大的元素向后移
            j--;
        }
        R[j + 1] = K; // 找到合适位置，插入元素
    }
}

// 优化的快速排序 - 能够自动切换到堆排序
void QuickSort(int R[], int m, int n, int max_depth, int current_depth) {
    extern int threshold; // 外部定义的阈值，小数组用插入排序
    int k;
    // 如果子数组很小，就先不排序，等最后统一用插入排序
    if (n - m + 1 <= threshold) {
        return;
    }

    // 如果递归太深（可能导致最坏情况），就改用堆排序
    if (current_depth >= max_depth) {
        HeapSort(R, m, n);
        return;
    }

    // 正常的快速排序流程
    if (m < n) {
        k = Partition(R, m, n); // 分区
        // 递归排序左右两部分
        QuickSort(R, m, k - 1, max_depth, current_depth + 1);
        QuickSort(R, k + 1, n, max_depth, current_depth + 1);
    }
}

// 主排序函数 - 整个排序过程的入口
void sort(int R[], int n) {
    // 计算最大递归深度限制 = 2*log2(n)
    int depth_limit = (int)(2 * log(n) / log(2));
    int i;

    printf("depth_limit:%d\n", depth_limit);

    // 先使用快速排序（可能会在某些情况下转为堆排序）
    QuickSort(R, 1, n, depth_limit, 0);

    // 打印快速排序后的中间结果
    printf("Intermediate:");
    for (i = 1; i <= n; i++) {
        printf("%d ", R[i]);
    }
    printf("\n");
    
    // 最后进行一次插入排序，完成整个数组的排序
    InsertionSort(R, n);
}
算法的智慧
这个排序算法很聪明，它结合了三种排序方法的优点：

快速排序：一般情况下速度很快
堆排序：在快速排序可能变慢的情况下接管排序
插入排序：对接近有序的小数组效率最高
就像煮饭，米饭先用大火煮开（快速排序），担心煮糊了就转中火（堆排序），最后用小火焖一会儿（插入排序），这样做出来的米饭又好吃又省时间！
