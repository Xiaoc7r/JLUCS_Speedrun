// AVL树节点结构
typedef struct AVLnode {
    int key;
    int height;
    struct AVLnode* left, * right;
} AVLnode;
typedef struct RotationInfo {
    int key;
    int level;
    char type[50];
} RotationInfo;
RotationInfo AA[100];
int rotation_count = 0;

AVLnode* newNode(int K) {
    AVLnode* node = (AVLnode*)malloc(sizeof(AVLnode));
    node->key = K;
    node->height = 0;
    node->left = node->right = NULL;
    return node;
}



int Height(AVLnode* node) {
    return node ? node->height : -1;
}
int max(int a, int b) {
    return a > b ? a : b;
}
void UpdateHeight(AVLnode* node) {
    if (node) node->height = max(Height(node->left), Height(node->right)) + 1;
}



void LL(AVLnode** node) {
    AVLnode* left = (*node)->left;
    (*node)->left = left->right;
    left->right = *node;
    UpdateHeight(*node);
    UpdateHeight(left);
    *node = left;
}
void RR(AVLnode** node) {
    AVLnode* right = (*node)->right;
    (*node)->right = right->left;
    right->left = *node;
    UpdateHeight(*node);
    UpdateHeight(right);
    *node = right;
}
void LR(AVLnode** node) {
    RR(&((*node)->left));
    LL(node);
}
void RL(AVLnode** node) {
    LL(&((*node)->right));
    RR(node);
}
