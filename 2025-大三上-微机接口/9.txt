A8255 EQU 0600H
B8255 EQU 0602H
C8255 EQU 0604H
M8255 EQU 0606H
DATA SEGMENT
TAB:				;TAB表用于记录0 ~ F的段选码
	DB 3FH,06H,5BH,4FH 
	DB 66H,6DH,7DH,07H 
	DB 7FH,6FH,77H,7CH 
	DB 39H,5EH,79H,71H
T1	DB 00H			;Ti表用于记录6个数码管的段选码
T2	DB 00H
T3	DB 00H
T4	DB 00H
T5	DB 00H
T6	DB 00H
FLAG DB 00H			;FLAG变量用于互斥操作
DATA ENDS
CODE SEGMENT
	ASSUME CS:CODE,DS:DATA
START:
	MOV AX,DATA		;导入DATA
	MOV DS,AX
	
	MOV AX,00H		;寄存器A清零
	LEA BX,TAB		;记录TAB表首地址
	
	MOV DX,M8255	;设置8255为A,B口输出，C口输入
	MOV AL,81H
	OUT DX,AL
	
MAIN:
	MOV AL,11110111B	;需要扫描的列，输出0
	MOV CX,04H			;一共需要扫描4列
M1:
	MOV DX,A8255
	OUT DX,AL			;从X口输出列选信号
	SHR AL,1			;右移一位
	OR AL,11110000B
	
	PUSH AX			;保存列选信号
	PUSH CX			;保存循环计数
	MOV DX,C8255
	IN AL,DX			;从Y口输入键盘反馈信号
	AND AL,0FH			;Y口接8255的C端口低四位，去除多余信息
	CMP AL,0FH			;如果低四位是1111，表明没有按键按下
	JE M2
	CALL SHOW		;没有跳去M2，则表明有按键按下
	JMP M3		;如果有按键按下，则跳过M2（互斥计数）阶段
M2:
	CMP FLAG,00H	;维持互斥信号，FLAG=MAX(0,FLAG-1)
	JE M3			;FLAG=0表示4列都没有按键按下，FLAG初值为4
	DEC FLAG		;每扫描一列，并且没有按键按下时，FLAG自减1次
M3:
	CALL CLEAR		;显示数码管的内容
	POP CX		;先取出循环计数
	POP AX		;再取出列选信号
	
	CALL DELAY	;稍微延迟
	LOOP M1
	
	JMP MAIN
	
SHOW:
	NOT AL			;NOT取反（从这里开始，通过Y口输入信息来判断按下的按键）
	AND AX,0FH		;AND取低4位（详情见下边第4点"关于按键判断"）
	CMP AL,01H	;0001
		JE D1
	CMP AL,02H	;0010
		JE D2
	CMP AL,04H	;0100
		JE D3
	CMP AL,08H	;1000
		JE D4
D1: MOV AL,04H
	JMP D5
D2: MOV AL,08H
	JMP D5
D3: MOV AL,0CH
	JMP D5
D4: MOV AL,10H
	JMP D5
D5: ADD AL,CL
	SUB AL,05H
	MOV SI,AX			;到这里，按下的按键被转化为独一无二的索引值
	CMP FLAG,00H		;是否存在互斥，非0表示存在互斥
	JNE DFI
	
	LEA BX,TAB		;把TAB的首地址放到BX
	MOV AL,T5		;将T5的值放到T6
	MOV T6,AL
	MOV AL,T4		;将T4的值放到T5
	MOV T5,AL
	MOV AL,T3		;将T3的值放到T4
	MOV T4,AL
	MOV AL,T2		;将T2的值放到T3
	MOV T3,AL
	MOV AL,T1		;将T1的值放到T2
	MOV T2,AL
	MOV AX,[BX+SI]		;将按键索引值放到T1
	MOV T1,AL			;T1是最右端的数码管
	
	CALL DELAY
DFI:
	MOV FLAG,04H		;FLAG初值设置
	RET
	
CLEAR:
	LEA BX,T1			;以T1为首地址，索引T2~T6
	MOV AL,11011111B
	MOV SI,00H
	MOV CX,06H			;数码管显示，与实验八相同
	MC2:
		MOV DX,A8255
		OUT DX,AL
		SHR AL,1
		OR AL,11000000B
		PUSH AX			;保存AX的值
			MOV DX,B8255
			MOV AL,[BX+SI]
			OUT DX,AL
		POP AX			;取出AX的值
		INC SI			;地址偏移自增
		CALL DELAY
	LOOP MC2
	RET
 
DELAY:
	PUSH BX
	MOV BX, 03FFH
	DEL:
		DEC BX
		JNZ DEL
	POP BX
	RET
CODE ENDS
	END START

; ====================== 区块1：硬件地址定义区【基础硬件配置，实验箱适配核心】 ======================
; 作用：给8255的A/B/C口和方式控制口分配固定操作地址，是程序控制键盘/数码管的“门牌号”，需与实验箱实际接线一致
; 【每一行可灵活修改的方式+具体效果（实验箱适配/功能调整，考试几乎不考此部分修改，硬件接线固定）】：
; 1. A8255 EQU 0600H → 8255 A口：输出键盘列选信号+数码管位选信号，若实验箱位选/列选接B口，需改此值为B8255，并同步互换后续A/B口的DX输出地址
; 2. B8255 EQU 0602H → 8255 B口：输出数码管段码信号，若实验箱段码接A口，需改此值为A8255，并同步互换后续A/B口的DX输出地址
; 3. C8255 EQU 0604H → 8255 C口：输入键盘行信号（低4位），若实验箱行信号接C口高4位，需改此值并同步修改行输入的AND AL,0FH为AND AL,F0H
; 4. M8255 EQU 0606H → 8255方式控制口，改此值后需同步修改初始化的DX值
A8255 EQU 0600H   ; 8255 A口：键盘列选+数码管位选输出（核心，硬件固定则无需改）
B8255 EQU 0602H   ; 8255 B口：数码管段码输出（核心，硬件固定则无需改）
C8255 EQU 0604H   ; 8255 C口：键盘行信号输入（低4位）
M8255 EQU 0606H   ; 8255方式控制口（硬件固定则无需改）

; ====================== 区块2：数据段定义区【键盘/数码管核心数据，考试改分关键！】 ======================
; 作用：存储**0-F的共阴极段码表、6个数码管的段码缓存、按键防抖动标志**，是键盘键值匹配和数码管显示的“数据源”，**考试90%的变种都改此区域**！
; 【每一行可灵活修改的方式+具体效果（覆盖所有显示/键盘变种，重点！）】：
; 一、段码表TAB（核心中的核心，考题必改！）：
;    共阴极段码对照表（题目已给）：0-3FH、1-06H、2-5BH、3-4FH、4-66H、5-6DH、6-7DH、7-07H、8-7FH、9-6FH、A-77H、B-7CH、C-39H、D-5EH、E-79H、F-71H
;    变种修改1：改为共阳极段码 → TAB表每个值取反（如0=0C0H、1=0F9H），并同步修改CLEAR子程序的段码输出逻辑
;    变种修改2：显示数字加小数点 → 对应段码+80H（如0带点=3FH|80H=BFH、1带点=06H|80H=86H）
;    变种修改3：仅显示0-9十进制 → 删除TAB表中A-F的段码（最后4行），并同步修改键盘键值映射为0-9
;    变种修改4：自定义显示字符 → 替换TAB表中的段码（如显示横线=40H、显示P=73H）
; 二、数码管段码缓存T1-T6（控制显示位置，核心！）：
;    T1 DB 00H → 最右侧数码管段码缓存（初始00H=熄灭），若改数码管位数（如4位），需删除T5/T6，同步修改CLEAR子程序的CX=06H为CX=04H
;    T2-T6 DB 00H → 从右到左第2-6位数码管段码缓存，初始均为00H（熄灭），变量名可改（如BUF1-BUF6），仅提升可读性
; 三、防抖动标志FLAG DB 00H：
;    1. 初始值00H → 改04H可调整防抖动次数（04H对应4列扫描，改03H则3列扫描防抖动）
;    2. 变量名可改（如KEY_FLAG DB 00H），无功能影响
DATA SEGMENT
TAB:				; 0~F的共阴极段码表（考题必改！核心中的核心！）
	DB 3FH,06H,5BH,4FH  ; 0、1、2、3的段码（改此值换显示字符，如加小数点→3FH|80H）
	DB 66H,6DH,7DH,07H  ; 4、5、6、7的段码
	DB 7FH,6FH,77H,7CH  ; 8、9、A、B的段码
	DB 39H,5EH,79H,71H  ; C、D、E、F的段码
T1	DB 00H			; 最右侧数码管段码缓存（初始熄灭，改初始值可默认显示数字）
T2	DB 00H			; 右2位数码管段码缓存
T3	DB 00H			; 右3位数码管段码缓存
T4	DB 00H			; 右4位数码管段码缓存
T5	DB 00H			; 右5位数码管段码缓存
T6	DB 00H			; 最左侧数码管段码缓存（6位改4位则删除T5/T6）
FLAG DB 00H			; 按键防抖动互斥标志（改初始值04H可调整防抖动次数）
DATA ENDS

; ====================== 区块3：代码段声明区【汇编程序的固定格式，不可乱改】 ======================
; 作用：告诉编译器代码段的名称、寄存器的归属关系（CS对应代码段，DS对应数据段），是汇编程序的“骨架”
; 【可灵活修改的方式+效果（仅可读性调整，无功能影响）】：
; 1. CODE SEGMENT → 段名可改（如MAIN_CODE SEGMENT），仅提升可读性
; 2. ASSUME CS:CODE,DS:DATA → 若数据段名改了（如DATA_SEG），需同步改DS:DATA_SEG，否则汇编报错
CODE SEGMENT
	ASSUME CS:CODE,DS:DATA

; ====================== 区块4：初始化区【程序启动配置，硬件固定则无需改】 ======================
; 作用：绑定数据段（访问TAB表/T1-T6）、初始化寄存器、配置8255工作模式（A/B口输出、C口输入），是程序运行的“准备工作”
; 【每一行可灵活修改的方式+效果（8255模式调整，考试低频变种）】：
; 1. MOV AX,DATA/MOV DS,AX → 绑定数据段，不可删除，否则程序无法访问TAB表/T1-T6
; 2. MOV AX,00H → 寄存器AX清零，无功能影响，可删除
; 3. LEA BX,TAB → BX指向段码表首地址，若段码表名改了（如SEG_TABLE），需同步改LEA BX,SEG_TABLE
; 4. MOV DX,M8255/MOV AL,81H/OUT DX,AL → 81H=8255方式控制字，**改此值可换输入输出模式**：
;    - 81H=10000001B → A口输出、B口输出、C口输入（本程序用此，匹配键盘/数码管接线）
;    - 80H=10000000B → A/B/C口均输出，需同步修改C口的行输入为输出，不可用
;    - 89H=10001001B → A/B口输出、C口输入，与81H功能一致
START:
	MOV AX,DATA		; 加载数据段地址（固定指令，不可改）
	MOV DS,AX		; DS绑定到数据段（固定指令，不可改）
	
	MOV AX,00H		; AX寄存器清零（无功能影响，可删除）
	LEA BX,TAB		; BX指向TAB段码表首地址（段码表名改则同步改）
	
	MOV DX,M8255	; 指向8255方式控制口（硬件固定则无需改）
	MOV AL,81H		; 8255控制字：A/B口输出、C口输入（改89H功能一致）
	OUT DX,AL		; 写入8255控制字（固定指令，不可改）

; ====================== 区块5：主循环键盘扫描区【键盘扫描核心，考试键盘变种改此部分！】 ======================
; 作用：循环扫描4列键盘，输出列选信号到8255 A口，读取C口的行输入信号，判断是否有按键按下，是键盘键值识别的“核心逻辑”
; 【每一行可灵活修改的方式+具体效果（键盘列数/扫描逻辑调整，考试高频变种！）】：
; 1. MOV AL,11110111B → 初始列选信号（第1列输出0，其余列1），改11111011B则从第2列开始扫描
; 2. MOV CX,04H → 扫描列数（4列），**改此值可换键盘列数，核心！**：
;    - 改CX=03H → 扫描3列键盘，需同步修改SHOW子程序的键值映射和FLAG的初值04H为03H
;    - 改CX=02H → 扫描2列键盘，需同步修改上述两处
; 3. MOV DX,A8255/OUT DX,AL → 输出列选信号到A口，若列选接B口，需改DX=B8255
; 4. SHR AL,1/OR AL,11110000B → 列选信号右移（切换到下一列），改SHL AL,1则从最后一列向前扫描
; 5. PUSH AX/PUSH CX → 保护列选信号和循环计数，不可删除，否则寄存器值丢失
; 6. MOV DX,C8255/IN AL,DX → 读取行输入信号，若行信号接C口高4位，需改AND AL,0FH为AND AL,F0H
; 7. AND AL,0FH → 保留C口低4位（行信号），改AND AL,F0H则保留高4位（适配行信号接线）
; 8. CMP AL,0FH → 判断是否有按键按下（0FH=无按键），改00H则逻辑反转（仅识别无按键）
; 9. JE M2 → 无按键则跳转到M2（防抖动），改JNZ则有按键跳转到M2
; 10. CALL SHOW → 有按键则调用SHOW子程序处理键值，不可删除
; 11. JMP M3 → 跳过M2的防抖动计数，不可删除
; 12. CMP FLAG,00H/JE M3/DEC FLAG → 防抖动逻辑（FLAG=0则无按键，否则减1），改FLAG=03H则3次扫描防抖动
; 13. CALL CLEAR → 调用数码管显示子程序，不可删除
; 14. POP CX/POP AX → 恢复循环计数和列选信号，不可删除
; 15. CALL DELAY → 调用短延时，改DELAY的BX值可调整扫描刷新率
; 16. LOOP M1/JMP MAIN → 循环扫描4列，无限主循环，不可删除
MAIN:
	MOV AL,11110111B	; 初始列选信号：第1列输出0（改此值换扫描起始列）
	MOV CX,04H			; 扫描列数：4列（改此值换键盘列数，核心！如3列→03H）
M1:
	MOV DX,A8255
	OUT DX,AL			; 输出列选信号到A口（列选接B口则改DX=B8255）

; ====================== 区块6：按键处理SHOW子程序区【键值映射+数字左移核心，考试移动变种改此部分！】 ======================
; 作用：将键盘的行列信号转化为0-F的十六进制索引，实现“数字左移、新键值显示在最右侧”的逻辑，是本实验的**核心功能区**！
; 【每一行可灵活修改的方式+具体效果（键值映射/移动方向调整，考试最高频变种！）】：
; 一、键值映射部分（D1-D5）：
; 1. NOT AL/AND AX,0FH → 行信号取反+保留低4位，不可修改
; 2. CMP AL,01H/JE D1 → 行1触发（0001），改02H则行2触发，适配键盘行接线
; 3. D1:MOV AL,04H → 行1对应键值偏移04H，改00H则重新映射键值（如行1对应0-3）
;    D2:MOV AL,08H → 行2对应键值偏移08H，改04H则重新映射
;    D3:MOV AL,0CH → 行3对应键值偏移0CH，改08H则重新映射
;    D4:MOV AL,10H → 行4对应键值偏移10H，改0CH则重新映射
; 4. ADD AL,CL/SUB AL,05H → 列数CL叠加偏移，转化为0-F的索引，改SUB AL,04H则重新映射键值
; 5. MOV SI,AX → 索引值存入SI，不可修改
; 二、数字移动部分（FLAG判断后）：
; 1. LEA BX,TAB → 重新指向段码表，不可修改
; 2. MOV AL,T5/MOV T6,AL → T5→T6（最左侧），是**左移的核心步骤**，**改赋值顺序可换移动方向，核心！**：
;    - 左移改右移 → 改为T2→T1、T3→T2…T6→T5，新键值存入T6（最左侧），如：
;      MOV AL,T2/MOV T1,AL; MOV AL,T3/MOV T2,AL; ... MOV AL,T6/MOV T5,AL; MOV AX,[BX+SI]/MOV T6,AL
;    - 不移出最左侧 → 删除T6的赋值，改为T1→T2…T4→T5，T5保持不变
; 3. MOV AX,[BX+SI]/MOV T1,AL → 新键值存入T1（最右侧），改T6则存入最左侧，适配显示需求
; 4. MOV FLAG,04H → FLAG初值设为04H（防抖动），改03H则3次防抖动
SHOW:

	AND AX,0FH		; 保留低4位（不可改）

		JE D1
	CMP AL,02H	; 行2触发（0010）
		JE D2
	CMP AL,04H	; 行3触发（0100）
		JE D3
	CMP AL,08H	; 行4触发（1000）
		JE D4
D1: MOV AL,04H		; 行1键值偏移（改此值重新映射键值，如00H→行1对应0-3）
	JMP D5
D2: MOV AL,08H		; 行2键值偏移（改此值重新映射键值）
	JMP D5
D3: MOV AL,0CH		; 行3键值偏移（改此值重新映射键值）
	JMP D5
D4: MOV AL,10H		; 行4键值偏移（改此值重新映射键值）
	JMP D5
D5: ADD AL,CL		; 叠加列数CL，转化为索引（不可改）
	SUB AL,05H		; 索引校准（改此值重新映射键值，如04H）
	MOV SI,AX			; 键值索引存入SI（不可改）
	CMP FLAG,00H		; 防抖动判断（不可改）
	JNE DFI			; 有互斥则跳过移动逻辑（不可改）
	
	LEA BX,TAB		; BX重新指向段码表（不可改）
	MOV AL,T5		; T5→T6（最左侧）（改赋值顺序换移动方向，核心！）
	MOV T6,AL
	MOV AL,T4		; T4→T5
	MOV T5,AL
	MOV AL,T3		; T3→T4
	MOV T4,AL
	MOV AL,T2		; T2→T3
	MOV T3,AL
	MOV AL,T1		; T1→T2
	MOV T2,AL
	MOV AX,[BX+SI]		; 读取新键值的段码（不可改）
	MOV T1,AL			; 新键值存入T1（最右侧）（改T6则存入最左侧）
	
	CALL DELAY		; 短延时（不可改）
DFI:

; ====================== 区块7：数码管显示CLEAR子程序区【动态扫描核心，数码管位数变种改此部分！】 ======================

    LEVEL DB 01H        ; 移动方向标志（01H=左移，02H=右移，实验核心的方向控制）
                        ; 可转换：改变量初值→LEVEL DB 02H→初始为右移；改变量值→03H→自定义新方向逻辑
DATA ENDS

; ===================== 区块3：代码段声明+程序初始化块【程序启动的准备工作，固定逻辑为主】 =====================
CODE SEGMENT
    ASSUME CS:CODE,DS:DATA  ; 绑定代码段(CS)和数据段(DS)
                            ; 可转换：改DATA为DATA_SEG→若数据段名改为DATA_SEG，需同步所有DATA引用
START:                      ; 程序入口标号（不可删，改名字需同步最后END的标号）
                            ; 可转换：改标号为MAIN_ENTRY→需同步最后一行END START改为END MAIN_ENTRY
    MOV AX,DATA             ; 将数据段地址送入AX寄存器（固定操作）
    ; 原代码笔误：MOV DX,AX → 修正为MOV DS,AX（绑定数据段，否则程序无法访问DATA中的变量）
    MOV DS,AX               ; 让DS寄存器指向数据段（核心修正，必须改！）
    MOV AX,00H              ; AX寄存器置0（初始化通用寄存器，无实际逻辑影响）
                            ; 可转换：改MOV AX,00H为MOV BX,00H→仅寄存器替换，功能不变
    LEA BX,TAB              ; BX寄存器指向段码表TAB的首地址（后续读取段码用）
                            ; 可转换：改BX为SI→仅寄存器替换；改TAB为TAB2→若段码表名改为TAB2
    MOV DX,CONTROL_8255     ; DX指向8255的控制口（准备配置工作模式）
    MOV AL,81H              ; 8255的工作模式字：A口输出、B口输出、C口输入（适配数码管+键盘）
                            ; 可转换：改81H为80H→C口输出（无键盘时用）；改91H→A口输入（位选接C口）
    OUT DX,AL               ; 将工作模式字写入8255控制口（固定操作，完成硬件初始化）

; ===================== 区块4：主循环块【键盘扫描+数码管显示的核心流程，改这里控制扫描/显示逻辑】 =====================
MAIN:
    MOV AL,11110111B        ; 键盘列选初始值：选择最左列（4列键盘，仅最低4位有效）
                            ; 可转换：改11110111B为11111011B→从第二列开始扫描；改11111110B→从第四列开始扫描
    MOV CX,04H              ; 键盘列扫描次数：4次（对应4列键盘）
                            ; 可转换：改04H为03H→扫描3列（适配3列键盘）；改05H→扫描5列（扩展键盘）
M1:                         ; 列扫描循环体标号
    MOV DX,A_8255           ; DX指向8255的A口（准备输出列选信号）
    OUT DX,AL               ; 将列选信号输出到A口（选中当前列）
    SHR AL,1                ; 列选信号右移一位（扫描下一列，最高位补0）
                            ; 可转换：改SHR AL,1为SHL AL,1→列选信号左移（反向扫描列）；改SHR AL,2→每次跳过一列扫描
    OR AL,11110000B         ; 保留高4位为1，仅低4位参与列选（防止高4位干扰）
    PUSH AX                 ; 压栈保存当前列选信号（后续恢复用）
    PUSH CX                 ; 压栈保存扫描次数（后续恢复用）
    MOV DX,C_8255           ; DX指向8255的C口（准备读取键盘行选反馈）
    IN AL,DX                ; 从C口读取键盘行选信号（有无按键的核心判断）
    AND AL,0FH              ; 仅保留低4位（行选信号只占低4位，高4位无效）
                            ; 可转换：改0FH为07H→仅读取低3位（适配3行键盘）
    CMP AL,0FH              ; 判断是否有按键：0FH=无按键（4行都为1），非0FH=有按键
                            ; 可转换：改0FH为07H→适配3行键盘的无按键判断
    JE M2                   ; 无按键则跳转到M2（跳过显示处理）
    CALL SHOW               ; 有按键则调用SHOW子程序（处理按键→数码管赋值+移动）
    JMP M3                  ; 跳转到M3（跳过无按键的互斥处理）
M2:                         ; 无按键处理标号
M3:                         ; 互斥标志处理标号
    CMP FLAG,00H            ; 判断互斥标志是否为0（0=无互斥，非0=需递减）
    JE M3                   ; 若为0则循环等待（直到FLAG非0）
    DEC FLAG                ; 互斥标志递减（FLAG=MAX(0,FLAG-1)，防止重复触发）
                            ; 可转换：改DEC FLAG为SUB FLAG,2→互斥次数减少2次，加快按键响应
    CALL CLEAR              ; 调用CLEAR子程序（刷新6位数码管显示）
    POP CX                  ; 出栈恢复扫描次数
    POP AX                  ; 出栈恢复列选信号
    CALL DELAY              ; 调用延时子程序（防止扫描过快导致的按键误判）
    LOOP M1                 ; CX减1，非0则跳回M1（继续扫描下一列）
    JMP MAIN                ; 跳回MAIN（无限循环，持续扫描键盘+显示数码管）

; ===================== 区块5：SHOW子程序块【按键处理+数码管移动的核心，实验需求的核心逻辑！】 =====================
; 功能：将按键转换为数字段码，按“左移→最左固定→右移”的规则更新数码管缓存
SHOW:
    NOT AL                  ; 对行选信号取反（无按键=0FH→00H，有按键=非0FH→对应行号）
    AND AX,0FH              ; 仅保留低4位（清除高12位干扰）
    ; 按键行号判断：将行选信号转换为行索引（01H=行1，02H=行2，04H=行3，08H=行4）
    CMP AL,01H
    JE D1
    CMP AL,02H
    JE D2
    CMP AL,04H
    JE D3
    CMP AL,08H
    JE D4
; 行索引转换为列偏移值（4列键盘，列偏移=04H/08H/0CH/10H）
D1:
    MOV AL,04H              ; 行1对应列偏移04H
                            ; 可转换：改04H为00H→行1对应列偏移00H（适配不同键盘接线）
    JMP D5
D2:
    MOV AL,08H              ; 行2对应列偏移08H
    JMP D5
D3:
    MOV AL,0CH              ; 行3对应列偏移0CH
    JMP D5
D4:
    MOV AL,10H              ; 行4对应列偏移10H
    JMP D5
D5:
    ADD AL,CL               ; 列偏移+列数（CL=CX，当前扫描列）→按键的全局索引
    SUB AL,05H              ; 索引修正（减去偏移，得到0~F的连续索引）
    MOV SI,AX               ; SI保存按键的段码表索引（0=0，1=1…15=F）
                            ; 可转换：改SI为DI→仅寄存器替换；改SUB AL,05H为SUB AL,04H→索引修正值适配不同键盘接线
    CMP FLAG,00H            ; 判断互斥标志（0=无互斥，非0=按键重复触发）
    JNE LL2                 ; 有互斥则跳转到LL2（跳过移动逻辑）
    LEA BX,TAB              ; 重新绑定BX到段码表首地址（防止BX被修改）
    CMP LEVEL,02H           ; 判断移动方向：02H=右移，01H=左移
    JE LEVEL_2              ; 右移则跳转到LEVEL_2

; LEVEL_1：左移逻辑（实验要求：第一次按键显最右端，后续左移）
LEVEL_1:
    MOV LEVEL,01H           ; 确认方向为左移（防止标志被修改）
    CMP TMP,06H             ; 判断是否到最左端（TMP=6=T6）
    JE LEVEL_2              ; 到最左端则切换为右移逻辑
    ; 数码管段码左移：T5→T6、T4→T5、T3→T4、T2→T3、T1→T2，新按键→T1
    MOV AL,T5               ; T5的值送入AL
    MOV T6,AL               ; AL的值写入T6（T5→T6）
    MOV AL,T4               ; T4的值送入AL
    MOV T5,AL               ; AL的值写入T5（T4→T5）
    MOV AL,T3
    MOV T4,AL
    MOV AL,T2
    MOV T3,AL
    MOV AL,T1
    MOV T2,AL
    MOV AX,[BX+SI]          ; 从段码表读取按键对应的段码
    MOV T1,AL               ; 新段码写入T1（最右端）
                            ; 可转换：改MOV T1,AL为MOV T2,AL→新按键写入右2位（修改左移的起始位）
    INC TMP                 ; 显示位置计数+1（向最左端移动）
                            ; 可转换：改INC TMP为ADD TMP,2→每次移动2位（跳过一位）
    JMP DEL                 ; 跳转到延时处理

; LEVEL_2：右移逻辑（实验要求：到最左端后，先固定最左，再右移）
LEVEL_2:
    MOV LEVEL,02H           ; 确认方向为右移
    CMP TMP,00H             ; 判断是否到最右端（TMP=0=T1）
    JE LEVEL_1              ; 到最右端则切换为左移逻辑
    CMP TMP,06H             ; 判断是否是最左端的第一次按键（TMP=6=T6）
    JE LL2                  ; 是则跳转到LL2（固定最左端显示，不执行右移）
    ; 数码管段码右移：T2→T1、T3→T2、T4→T3、T5→T4、T6→T5，新按键→T6
    MOV AL,T2
    MOV T1,AL
    MOV AL,T3
    MOV T2,AL
    MOV AL,T4
    MOV T3,AL
    MOV AL,T5
    MOV T4,AL
    MOV AL,T6
    MOV T5,AL
                            ; 可转换：删除上述右移指令→取消右移，仅固定最左端显示
LL2:
    CMP FLAG,00H            ; 再次判断互斥标志（防止重复赋值）
    JNE DFI                 ; 有互斥则跳转到DFI
    MOV AX,[BX+SI]          ; 读取按键对应的段码
    MOV T6,AL               ; 新段码写入T6（最左端，实现“最左固定”的实验要求）
                            ; 可转换：改MOV T6,AL为MOV T5,AL→新按键写入左2位（修改右移的起始位）
    DEC TMP                 ; 显示位置计数-1（向最右端移动）
                            ; 可转换：改DEC TMP为SUB TMP,2→每次移动2位（跳过一位）
DEL:
    CALL DELAY              ; 调用延时子程序（防止按键抖动）
DFI:
    MOV FLAG,04H            ; 设置互斥标志初值：4次扫描内不重复触发（防抖动）
                            ; 可转换：改04H为02H→互斥次数为2，加快按键响应；改08H→互斥次数为8，更防抖动
    RET                     ; 子程序返回

; ===================== 区块6：CLEAR子程序块【数码管刷新显示的执行单元，改这里控制显示位数/顺序】 =====================
; 功能：按T1~T6的缓存刷新6位数码管，检测到F键则退出程序
CLEAR:
    LEA BX,T1               ; BX指向最右端数码管的段码缓存T1（后续按索引读取T1~T6）
                            ; 可转换：改BX为SI→仅寄存器替换；改LEA BX,T6→从最左端开始显示（反向显示）
    MOV AL,11011111B        ; 数码管位选初始值：选择最右端T1（低6位有效，1=熄灭，0=点亮）
                            ; 可转换：改11011111B为11111101B→从右2位T2开始显示；改10111111B→从最左端T6开始显示
    MOV SI,00H              ; 数码管缓存索引（0=T1，1=T2…5=T6）
    MOV CX,06H              ; 数码管显示次数：6次（对应6位显示，实验要求）
                            ; 可转换：改06H为05H→只显示5位（隐藏最左端T6）；改04H→只显示4位（隐藏T5/T6）
MC2:                        ; 数码管刷新循环体标号
    MOV DX,A_8255           ; DX指向8255的A口（输出位选信号）
    OUT DX,AL               ; 输出位选信号（点亮当前数码管）
    SHR AL,1                ; 位选信号右移一位（点亮下一个数码管）
                            ; 可转换：改SHR AL,1为SHL AL,1→反向点亮（从左到右）
    OR AL,11000000B         ; 保留高2位为1，仅低6位参与位选（防止干扰）
    PUSH AX                 ; 压栈保存当前位选信号
    MOV DX,B_8255           ; DX指向8255的B口（输出段选信号）
    MOV AL,[BX+SI]          ; 读取当前索引的数码管段码缓存（T1~T6）
    CMP AL,71H              ; 判断是否是F键的段码（71H）
                            ; 可转换：改71H为79H→按E键退出；改71H为3FH→按0键退出
    JE TUICHU               ; 是则跳转到TUICHU（退出程序）
    OUT DX,AL               ; 输出段选信号（当前数码管显示对应数字）
    POP AX                  ; 出栈恢复位选信号
    INC SI                  ; 索引+1（读取下一个数码管缓存）
    CALL DELAY              ; 延时（防止显示闪烁）
    LOOP MC2                ; CX减1，非0则跳回MC2（刷新下一个数码管）
    RET                     ; 子程序返回

; ===================== 区块7：退出程序块【实验要求的F键退出，固定逻辑】 =====================
TUICHU:
    MOV AH,4CH              ; DOS中断号：程序退出
    INT 21H                 ; 调用DOS中断（结束程序，固定操作）

; ===================== 区块8：延时子程序块【控制按键防抖/显示闪烁，改这里调延时长度！】 =====================
DELAY:
    PUSH CX                 ; 压栈保存CX（防止覆盖主程序的计数）
    MOV CX,03FFH            ; 延时计数初值：03FFH=1023次循环
                            ; 可转换：改03FFH为01FFH→延时更短（按键响应更快，可能闪烁）；改07FFH→延时更长（显示更稳定，按键响应慢）；改1000H→16位计数，延时极长
L1:
    LOOP $                  ; 空循环延时（$表示当前指令地址，固定操作）
    POP CX                  ; 出栈恢复CX
    RET                     ; 子程序返回

; ===================== 区块9：程序结束块【固定格式，不可删】 =====================
CODE ENDS
END START                   ; 程序结束，指定入口标号为START
                            ; 可转换：改START为MAIN_ENTRY→若入口标号修改，需同步
