; ====================== 区块1：硬件地址定义区 ======================
; 作用：给8255的A/B/C口和控制寄存器分配操作地址，是程序操作硬件的“门牌号”
; 【可灵活修改】：所有EQU后的十六进制数可改，适配不同实验箱的8255端口地址
A8255    EQU 0640H   ; 8255的A口：读取开关K7~K0的输入信号（改此值→换A口硬件地址）
B8255    EQU 0642H   ; 8255的B口：控制数据灯的输出显示（改此值→换B口硬件地址）
C8255    EQU 0644H   ; 8255的C口：本实验未使用，预留（改此值→换C口硬件地址）
MODE8255 EQU 0646H   ; 8255控制寄存器：配置工作模式（改此值→换控制寄存器地址）

; ====================== 区块2：代码段声明区 ======================
; 作用：汇编程序的固定格式，告诉编译器代码段的归属，无实际功能逻辑
; 【可灵活修改】：无（汇编固定语法，无需修改）
CODE SEGMENT
     ASSUME CS:CODE  ; 绑定代码段寄存器CS与CODE段

; ====================== 区块3：8255初始化区块 ======================
; 作用：配置8255的工作模式，让A口做输入、B口做输出（适配开关输入+数据灯输出的需求）
; 【可灵活修改】：AL=90H 是8255的控制字，修改此值可改A/B口的输入输出模式
START:
     MOV DX,MODE8255 ; 指向8255控制寄存器
     MOV AL,90H      ; 8255控制字：A口输入、B口输出（改90H→换工作模式，如80H=A口输出/B口输入）
     OUT DX,AL       ; 将控制字写入8255，完成初始化

; ====================== 区块4：主循环核心功能区 ======================
; 作用：循环读取A口开关状态，判断是否满足退出条件；不满足则将A口值输出到B口
; 【可灵活修改1】：CMP AL,0FFH → 修改0FFH可改退出条件（如改0F0H=K7~K4全高电平退出）
; 【可灵活修改2】：MOV DX,B8255 → 若改B8255为C8255，可将输出切换到C口（接其他外设）
MAIN:    
     MOV DX,A8255    ; 指向8255的A口
     IN AL,DX        ; 读取A口的开关K7~K0状态，存入AL
     
     CMP AL,0FFH     ; 判断是否全高电平（0FFH），是则跳转到EXIT退出
     JE EXIT         ; 满足退出条件，跳转到退出区块
     
     MOV DX,B8255    ; 指向8255的B口
     OUT DX,AL       ; 将A口的开关状态输出到B口，控制数据灯亮灭
     JMP MAIN        ; 无限循环，持续检测开关状态

; ====================== 区块5：程序退出功能区 ======================
; 作用：满足退出条件时，熄灭数据灯并调用DOS中断结束程序
; 【可灵活修改1】：MOV AL,00H → 修改00H可改退出时数据灯状态（如改0FFH=退出时数据灯全亮）
; 【可灵活修改2】：MOV AH,4CH → 4CH是DOS退出中断，固定值无需修改
EXIT:
     MOV DX,B8255    ; 指向8255的B口
     MOV AL,00H      ; 输出00H，熄灭所有数据灯（改此值→退出时灯的亮灭状态变化）
     OUT DX,AL       ; 将熄灭指令写入B口
     MOV AH,4CH      ; DOS中断号：程序正常退出
     INT 21H         ; 调用DOS中断，结束程序

; ====================== 区块6：代码段结束区 ======================
; 作用：汇编程序的固定格式，标记代码段结束并指定程序入口
; 【可灵活修改】：无（汇编固定语法，无需修改）
CODE ENDS
     END START       ; 代码段结束，指定程序入口为START标号

; 作用：循环读取A口，仅提取K0的电平，根据K0状态控制B口输出
; 【可灵活修改1】：AND AL,01H → 01H是检测K0的掩码，改此值可换检测的开关位（如02H=检测K1，04H=检测K2）
; 【可灵活修改2】：MOV AL,0FH → 0FH是低4位亮的控制码，改此值可改低几位亮（如03H=低2位亮，07H=低3位亮）

     MOV DX,A8255
     IN AL,DX        ; 读取A口的开关状态
     
     AND AL,01H      ; 掩码过滤，仅保留K0的电平（改01H→换检测的开关位）
     
     CMP AL,01H      ; 判断K0是否为高电平
     JE AA0          ; K0高电平，跳转到AA0（输出高4位亮）
     MOV AL,0FH      ; K0低电平，赋值0FH（低4位亮）（改此值→换低亮的灯位）
     JMP NEXT        ; 跳转到NEXT，输出到B口

     MOV AL,0F0H     ; K0高电平，赋值0F0H（高4位亮）（改此值→换高亮的灯位）

; ====================== 区块5：输出控制区 ======================
; 作用：将控制码输出到B口，控制数据灯亮灭
; 【可灵活修改】：MOV DX,B8255 → 改B8255为C8255，输出切换到C口
NEXT:     
     MOV DX,B8255

; ====================== 区块4：主循环核心功能区 ======================
; 作用：循环读取A口，提取K1K0的组合电平，根据四种组合赋值对应的控制码
; 【可灵活修改1】：AND AL,03H → 03H是检测K1K0的掩码，改此值可换检测的开关组合（如07H=检测K2K1K0，0CH=检测K3K2）
; 【可灵活修改2】：CMP AL后的00H/01H/02H → 改此值可换开关组合的判断条件（如改02H为04H=检测K2为1）
; 【可灵活修改3】：MOV AL后的00H/0FH/0F0H/0FFH → 改此值可换每种组合下灯的亮灭状态（如05H=低4位的第0、2位亮，0A0H=高4位的第1位亮）
MAIN:    
     MOV DX,A8255
     IN AL,DX        ; 读取A口的开关状态
     
     AND AL,03H      ; 掩码过滤，仅保留K1K0的电平（改03H→换检测的开关组合）
     
     CMP AL,00H      ; 判断K1K0是否为00
     JE AA0          ; 跳转到AA0（灯全灭）
     CMP AL,01H      ; 判断K1K0是否为01
     JE AA1          ; 跳转到AA1（低4位亮）
     CMP AL,02H      ; 判断K1K0是否为10
     JE AA2          ; 跳转到AA2（高4位亮）
     MOV AL,0FFH     ; K1K0=11，赋值0FFH（灯全亮）（改此值→换全亮的替代状态）
     JMP NEXT        ; 跳转到NEXT，输出到B口
AA0:
     MOV AL,00H      ; K1K0=00，赋值00H（灯全灭）（改此值→换全灭的替代状态）
     JMP NEXT
AA1:
     MOV AL,0FH      ; K1K0=01，赋值0FH（低4位亮）（改此值→换低亮的灯位）
     JMP NEXT
AA2:
     MOV AL,0F0H     ; K1K0=10，赋值0F0H（高4位亮）（改此值→换高亮的灯位）
