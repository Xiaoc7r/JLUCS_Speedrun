A8254 EQU 0640H          	;8254的计数器0地址 
M8254 EQU 0646H      		;8254的控制寄存器地址 
A8255 EQU 0600H				;8255相关地址
B8255 EQU 0602H
C8255 EQU 0604H
M8255 EQU 0606H
 
DATA SEGMENT
TAB:
	DB 3FH,06H,5BH,4FH 
	DB 66H,6DH,7DH,07H 
	DB 7FH,6FH,77H,7CH 
	DB 39H,5EH,79H,71H,00H
TAB2:
	T6	DB 10H
	T5	DB 10H
	T4	DB 10H
	T3	DB 10H
	T2	DB 10H
	T1	DB 10H
	
STATE 	DB 00H
FLAG 	DB 00H
DATA ENDS
 
CODE SEGMENT
	ASSUME CS:CODE,DS:DATA
START:
	MOV AX,DATA				;导入数据段的变量
	MOV DS,AX
	
SET8254: 
    MOV AL,36H      		;计时器 0 以方式 3 工作
    MOV DX,M8254
    OUT DX,AL      
    
    MOV DX,A8254			;初值设置4800H
    MOV AL,00H
    OUT DX,AL
    MOV AL,48H
    OUT DX,AL
    
SET8255:
	MOV DX,M8255			;AB输出，C输入
	MOV AL,81H
	OUT DX,AL
	
SET8259:
    MOV AX,OFFSET MIR6		;设置中断向量MIR6
    MOV SI,0038H
    MOV [ES:SI],AX			;由于导入了数据段，这里设置变量需要改用ES:SI的方式
    MOV AX,CS
    MOV SI,003AH
    MOV [ES:SI],AX
	
    CLI  		;关中断
    MOV AL, 11H    	;ICW1 
    OUT 20H, AL
    MOV AL, 08H		;ICW2
    OUT 21H, AL
    MOV AL, 04H		;ICW3
    OUT 21H, AL
    MOV AL, 07H 	;ICW4
    OUT 21H, AL
    MOV AL, 2FH  	;OCW1
    OUT 21H, AL
    STI  		;开中断
    
MAIN:
	MOV AL,11110111B	;键盘扫描
	MOV CX,04H
M1:
	MOV DX,A8255
	OUT DX,AL
	SHR AL,1
	
	PUSH AX				;保存寄存器内容
	PUSH CX
	MOV DX,C8255
	IN AL,DX
	AND AL,0FH
	CMP AL,0FH
	JE M2
	CALL JUDGE
	JMP M3
M2:
	CMP FLAG,00H
	JE M3
	DEC FLAG
M3:
	CALL SHOW
	POP CX				;还原寄存器内容
	POP AX
	
	CALL DELAY
	LOOP M1
	JMP MAIN
	
JUDGE:
	NOT AL
	AND AX,0FH
	
	CMP AL,01H			;SWITCH，解码键盘扫描
	JE D1
	CMP AL,02H
	JE D2
	CMP AL,04H
	JE D3
	CMP AL,08H
	JE D4
D1:
	MOV AL,04H
	JMP D5
D2:
	MOV AL,08H
	JMP D5
D3:
	MOV AL,0CH
	JMP D5
D4:
	MOV AL,10H
	JMP D5
D5:			
	ADD AL,CL
	SUB AL,05H
	MOV SI,AX			;0-F按键转换为00H-0FH的偏移SI
	
	CMP FLAG,00H		;按键互斥操作
	JNE JUDGE_END
	
	CMP SI,0AH			;A ~ F按键的if-else操作
		JNE D0A
		CALL PRESS_A
		JMP JUDGE_END
	D0A:
	CMP SI,0BH
		JNE D0B
		CALL PRESS_B
		JMP JUDGE_END
	D0B:
	CMP SI,0CH
		JNE D0C
		CALL PRESS_C
		JMP JUDGE_END
	D0C:
	CMP SI,0DH
		JNE D0D
		CALL PRESS_D
		JMP JUDGE_END
	D0D:
	CMP SI,0EH
		JNE D0E
		CALL PRESS_E
		JMP JUDGE_END
	D0E:
	CMP SI,0FH
		JNE D0F
		CALL PRESS_F
		JMP JUDGE_END
	D0F:
	
	CMP STATE,00H
	JNE JUDGE_END			;初始状态才能设置计时器初值
	
	LEA BX,TAB
	
	MOV AL,T2
	MOV T1,AL
	MOV AX,SI
	MOV T2,AL
	
	JUDGE_END:
		MOV FLAG,04H		;互斥
		RET
 
PRESS_A:
    CMP STATE,02H	;是否正在暂停，则回到初始状态
		JE D_4
	CMP STATE,01H	;是否正在工作，则回到初始状态
		JE D_4
	CMP STATE,00H	;是否非初始状态
		JNE D_END
		
    CMP T2,10H		;T2未设置
        JE D_END
    CMP T2,00H		;T2被设置为0，需要进一步判断T1是否设置
        JE D_1
    CMP T1,10H		;T2被设置为1到9，但T1未设置，需要将T1置零
        JE D_2
    CMP T1,00H
        JE D_3
	D_1:	
		CMP T1,10H		;T1未设置，无法开始
			JE D_END
		CMP T1,00H		;T1也被设置为0，无法开始
			JE D_END
		JMP D_3			;T1被设置为1到9，开始计时
	D_2:
		MOV T1,00H		;T1置零
		JMP D_3			;开始计时
	D_3:
		MOV STATE,01H	;修改状态
		MOV T3,00H		;设置 秒 显示
		MOV T4,00H
        JMP D_END		
	D_4:
        MOV STATE,00H	;修改状态
        MOV T1,10H		;设置 分和秒 不显示
        MOV T2,10H
        MOV T3,10H
        MOV T4,10H
        JMP D_END
	D_END:
		RET
PRESS_B:
    CMP STATE,01H		;如果正在计时，就暂停
		JE E1
	CMP STATE,02H		;如果正在暂停，就计时
		JE E2
	JMP E3
	E1:
		MOV STATE,02H	;修改为暂停状态
		JMP E3
	E2:
		MOV STATE,01H	;修改为计时状态
		JMP E3
	E3:
		RET
PRESS_C:
    CALL CLEAR_SHOW
    MOV STATE,00H	;修改状态
    MOV T1,10H		;设置 分和秒 不显示
    MOV T2,10H
    MOV T3,10H
    MOV T4,10H
	JMP EXIT		;退出程序
		RET
PRESS_D:				;开始倒计时
		RET
	
PRESS_E:
		RET
		
PRESS_F:
		RET
 
EXIT:
	MOV AH,4CH
	INT 21H
	
CLEAR_SHOW:				;清除显示（有更简单的写法）
	MOV AL,11011111B
	MOV CX,06H
	CS2:
		MOV DX,A8255
		OUT DX,AL
		SHR AL,1
		OR AL,11000000B
		
		PUSH AX
		MOV DX,B8255
		MOV AL,00H
		OUT DX,AL
		POP AX
		
		CALL DELAY
	LOOP CS2
	RET
	
SHOW:					;根据偏移显示对应数字
	MOV AL,11011111B
	
	MOV SI,00H
	MOV CX,06H
	S2:
		
		MOV DX,A8255
		OUT DX,AL
		
		SHR AL,1
		OR AL,11000000B
		
		PUSH AX
			LEA BX,TAB2
			MOV AX,00H
			MOV AL,[BX+SI]			;第一次偏移获取Ti的值
			PUSH SI
				LEA BX,TAB
				MOV SI,AX
				MOV AL,[BX+SI]		;第二次偏移获取0到9对应数码管段选码的值
				MOV DX,B8255
				OUT DX,AL
			POP SI
		POP AX
		
		INC SI
		CALL DELAY
	
	LOOP S2
	
	RET
	
MIR6:		;定时器触发的MIR6中断
	STI
	PUSH CX
	PUSH BX
	PUSH AX
	
	CMP STATE,01H		;如果不在计时状态，则结束
	JNE MIR_END
	
	CALL TIME_DEC		;-1
	JMP MIR_END
		
	MIR_END:
		POP AX
		POP BX
		POP CX
	IRET
 
TIME_DEC:
	CMP T4,00H			;借位减法
		JNE DEC_T4
		MOV T4,09H
	CMP T3,00H
		JNE DEC_T3
		MOV T3,05H
	CMP T2,00H
		JNE DEC_T2
		MOV T2,09H
	CMP T1,00H
		JNE DEC_T1
		
	CALL TIME_END		;借无可借
	JMP DEC_END
		
	DEC_T4:
		DEC T4
		JMP DEC_END
	DEC_T3:
		DEC T3
		JMP DEC_END
	DEC_T2:
		DEC T2
		JMP DEC_END
	DEC_T1:
		DEC T1
		
	DEC_END:
		RET
	
	
TIME_END:
	PUSH CX
	
	MOV STATE,03H		;计时结束状态
	MOV T4,10H
	MOV T3,10H
	MOV T2,10H
	MOV T1,10H
	
		MOV CX, 03H		;闪烁3次
	TIME_SHIN:
		CALL TIME_OUT0
		CALL DELAY_LONG
		CALL TIME_CLEAR
		CALL DELAY_LONG
		LOOP TIME_SHIN
		
	MOV STATE,00H		;回归初始状态
	POP CX
	RET
 
TIME_OUT0:			;输出0000
	PUSH AX
		MOV AL,11110000B
		MOV DX,A8255
		OUT DX,AL
		
		MOV AL,3FH
		MOV DX,B8255
		OUT DX,AL
	POP AX
	RET
	
TIME_CLEAR:			;清除显示
	PUSH AX
		MOV AL,11110000B
		MOV DX,A8255
		OUT DX,AL
		
		MOV AL,00H
		MOV DX,B8255
		OUT DX,AL
	POP AX
	RET
	
DELAY:
	PUSH BX
	MOV BX, 01FFH
	DEL:
		DEC BX
		JNZ DEL
	POP BX
	RET
	
DELAY_LONG:
	PUSH BX
	PUSH CX
	MOV BX, 0FFFFH
	DEL2:
		MOV CX, 06H
		DEL3:
			LOOP DEL3
		DEC BX
		JNZ DEL2
	POP CX
	POP BX
	RET
	
CODE ENDS
	END START

; ===================== 区块1：硬件地址定义块【8254/8255硬件交互的“门牌号”，程序通过这些地址控制硬件】 =====================
; 可转换点：改EQU后的数值→适配不同硬件地址；互换A8255/B8255→适配数码管位选/段选的硬件接线（硬件不变则无需改）
A8254 EQU 0640H           ; 8254的计数器0地址（产生1秒计时中断的核心计数器）
                          ; 可转换：改0641H→适配新的8254计数器0地址；改B8254→使用计数器1（需同步控制字）
M8254 EQU 0646H           ; 8254的控制寄存器地址（配置计数器工作模式）
                          ; 可转换：改0647H→适配新的8254控制口地址
A8255 EQU 0600H           ; 8255的A口地址（输出数码管位选/键盘列选信号）
                          ; 可转换：改0601H→适配新的A口地址；和B8255互换→位选接B口、段选接A口
B8255 EQU 0602H           ; 8255的B口地址（输出数码管段选信号）
                          ; 可转换：改0603H→适配新的B口地址；和A8255互换→段选接A口、位选接B口
C8255 EQU 0604H           ; 8255的C口地址（输入键盘行选反馈信号）
                          ; 可转换：改0605H→适配新的C口地址；无键盘则改不改不影响
M8255 EQU 0606H           ; 8255的控制寄存器地址（配置8255工作模式）
                          ; 可转换：改0607H→适配新的8255控制口地址

; ===================== 区块2：数据段定义块【存储核心数据，90%考题变种改这里！】 =====================
DATA SEGMENT
    TAB:                  ; 0~F的共阴极数码管段码表（含熄灭码00H，实验核心的显示数据源）
    DB 3FH,06H,5BH,4FH    ; 对应数字：0    1    2    3
    DB 66H,6DH,7DH,07H    ; 对应数字：4    5    6    7
    DB 7FH,6FH,77H,7CH    ; 对应数字：8    9    A    B
    DB 39H,5EH,79H,71H    ; 对应数字：C    D    E    F
    DB 00H                ; 数码管熄灭码（00H=无显示，实验初始状态用）
                          ; 可转换：改段码值→如3FH→BFH（显示0.带小数点）；删除熄灭码→用71H（F）作为熄灭；新增段码→扩展显示特殊字符（如横杠040H）
    TAB2:                 ; 6位数码管的段选码缓存（T1=最右端，T6=最左端；实验用T1-T4显分/秒，T5-T6预留）
    T6 DB 10H             ; 最左端数码管缓存（10H=无效值，对应熄灭码，实验未使用）
                          ; 可转换：改10H为00H→初始显示0；新增T7 DB 10H→扩展为7位显示
    T5 DB 10H             ; 左2位数码管缓存（实验未使用）
    T4 DB 10H             ; 右4位数码管缓存→秒的个位（实验核心显示位）
    T3 DB 10H             ; 右3位数码管缓存→秒的十位（实验核心显示位）
    T2 DB 10H             ; 右2位数码管缓存→分的个位（实验核心显示位）
    T1 DB 10H             ; 最右端数码管缓存→分的十位（实验核心显示位）
                          ; 可转换：互换T1/T4→分秒显示位置对调；改10H为01H→初始显示分十位为1
    STATE DB 00H          ; 系统状态标志（实验核心的逻辑控制变量）
                          ; 可转换：定义新状态值→04H=循环计时；05H=加计时（原实验为减计时）
                          ; 状态值定义：00H=初始状态（无显示/可输初值）、01H=计时中、02H=暂停中、03H=计时结束
    FLAG DB 00H           ; 按键互斥标志（防止按键重复触发，0=无互斥，非0=有互斥）
                          ; 可转换：改初值为04H→初始互斥；改互斥值为02H→加快按键响应；08H→更防抖动
DATA ENDS

; ===================== 区块3：代码段声明+程序入口【程序启动的准备工作，固定逻辑为主】 =====================
CODE SEGMENT
    ASSUME CS:CODE,DS:DATA  ; 绑定代码段(CS)和数据段(DS)
                            ; 可转换：改DATA为DATA_SEG→若数据段名改为DATA_SEG，需同步所有DATA引用
START:                      ; 程序入口标号（不可删，改名字需同步最后END的标号）
                            ; 可转换：改标号为MAIN_ENTRY→需同步最后一行END START改为END MAIN_ENTRY
    MOV AX,DATA             ; 将数据段地址送入AX寄存器（固定操作）
    MOV DS,AX               ; 让DS寄存器指向数据段（核心操作，访问DATA中的变量）
    ; 补充：原代码缺失ES段初始化（中断向量表在0段），需添加以下两行（否则MOV [ES:SI],AX会出错）
    MOV AX,0000H            ; ES指向中断向量表的0段（必须补充，否则中断设置失效）
    MOV ES,AX               ; 初始化附加段ES（修正原代码的语法疏漏）

; ===================== 区块4：8254初始化块【产生1秒计时中断的核心，改这里控制中断周期！】 =====================
SET8254:                    ; 8254计数器0初始化标号（产生1秒方波中断）
    MOV AL,36H              ; 8254计数器0的控制字：模式3（方波）、二进制计数、高低字节送数
                            ; 可转换：改36H为35H→模式2（速率发生器）；改76H→使用计数器1；改16H→模式1（单稳态）
    MOV DX,M8254            ; DX指向8254的控制寄存器
    OUT DX,AL               ; 将控制字写入8254（固定操作）
    
    MOV DX,A8254            ; DX指向8254的计数器0
    MOV AL,00H              ; 计数初值的低8位（整体初值=4800H，配合系统时钟产生1秒中断）
                            ; 可转换：改00H为50H→低8位为50H，中断周期缩短（如0.5秒）；改00H为FFH→周期延长
    OUT DX,AL               ; 写入低8位初值（固定操作）
    MOV AL,48H              ; 计数初值的高8位（4800H为实验预设的1秒初值）
                            ; 可转换：改48H为24H→高8位为24H，中断周期0.5秒；改96H→周期2秒
    OUT DX,AL               ; 写入高8位初值（固定操作）

; ===================== 区块5：8255初始化块【驱动数码管/扫描键盘的硬件配置，改这里控制显示/扫描模式】 =====================
SET8255:                    ; 8255初始化标号（A/B口输出，C口输入）
    MOV DX,M8255            ; DX指向8255的控制寄存器
    MOV AL,81H              ; 8255的工作模式字：A口输出（位选/列选）、B口输出（段选）、C口输入（行选）
                            ; 可转换：改81H为80H→C口输出（无键盘时用）；改91H→A口输入（位选接C口）
    OUT DX,AL               ; 将工作模式字写入8255（固定操作）

; ===================== 区块6：8259中断初始化块【管理8254的计时中断，改这里控制中断优先级/屏蔽】 =====================
SET8259:                    ; 8259中断控制器初始化标号（管理6号中断）
    MOV AX,OFFSET MIR6      ; 取中断服务程序MIR6的偏移地址
                            ; 可转换：改MIR6为MIR6_TIME→若服务程序名修改，需同步所有引用
    MOV SI,0038H            ; SI指向6号中断向量表的偏移地址位（6*4=24=18H？原代码为自定义6号中断，按原码来）
                            ; 可转换：改0018H→适配3号中断；改0020H→适配5号中断
    MOV [ES:SI],AX          ; 将MIR6的偏移地址写入中断向量表（修正后用ES段，原代码缺失ES初始化）
    MOV AX,CS               ; 取代码段(CS)的地址（固定操作）
    MOV SI,003AH            ; SI指向6号中断向量表的段地址位
                            ; 可转换：改001AH→适配3号中断；改0022H→适配5号中断
    MOV [ES:SI],AX          ; 将MIR6的段地址写入中断向量表（固定操作）
    
    CLI                     ; 关中断（初始化中断时防止干扰，固定操作）
    MOV AL,11H              ; 8259的ICW1：边沿触发、单片8259
                            ; 可转换：改13H→电平触发中断；改19H→多片8259配置
    OUT 20H,AL              ; 将ICW1写入8259主片的20H端口（固定操作）
    MOV AL,08H              ; 8259的ICW2：设置中断号基址为8
                            ; 可转换：改09H→中断号基址为9，适配不同中断优先级
    OUT 21H,AL              ; 将ICW2写入8259的21H端口（固定操作）
    MOV AL,04H              ; 8259的ICW3：主片的IR2接从片（本实验无片，按原码来）
                            ; 可转换：改02H→主片IR1接从片；改00H→无片连接
    OUT 21H,AL              ; 将ICW3写入8259的21H端口（固定操作）
    MOV AL,07H              ; 8259的ICW4：8086模式、普通结束中断
                            ; 可转换：改01H→自动结束中断；改05H→缓冲模式
    OUT 21H,AL              ; 将ICW4写入8259的21H端口（固定操作）
    MOV AL,2FH              ; 8259的OCW1：开放6号中断，屏蔽其他中断
                            ; 可转换：改2BH→屏蔽6号、开放5号；改2FH→仅开放6号（实验用）
    OUT 21H,AL              ; 将OCW1写入8259的21H端口（固定操作）
    STI                     ; 开中断（中断初始化完成，开启中断，固定操作）

; ===================== 区块7：主循环块【键盘扫描+数码管显示的核心流程，改这里控制扫描/显示逻辑】 =====================
MAIN:                       ; 主循环标号（无限扫描键盘+刷新显示）
    MOV AL,11110111B        ; 键盘列选初始值：选择最左列（4列键盘，仅最低4位有效）
                            ; 可转换：改11110111B为11111011B→从第二列开始扫描；改11111110B→从第四列开始扫描
    MOV CX,04H              ; 键盘列扫描次数：4次（对应4列键盘）
                            ; 可转换：改04H为03H→扫描3列（适配3列键盘）；改05H→扫描5列（扩展键盘）
M1:                         ; 列扫描循环体标号
    MOV DX,A8255            ; DX指向8255的A口（输出列选信号）
    OUT DX,AL               ; 将列选信号输出到A口（选中当前列）
    SHR AL,1                ; 列选信号右移一位（扫描下一列，最高位补0）
                            ; 可转换：改SHR AL,1为SHL AL,1→列选信号左移（反向扫描列）；改SHR AL,2→每次跳过一列扫描
    ; 原代码缺失：OR AL,11110000B（防止高4位干扰，补充后更稳定）
    PUSH AX                 ; 压栈保存当前列选信号（后续恢复用）
    PUSH CX                 ; 压栈保存扫描次数（后续恢复用）
    MOV DX,C8255            ; DX指向8255的C口（读取键盘行选反馈）
    IN AL,DX                ; 从C口读取行选信号（有无按键的核心判断）
    AND AL,0FH              ; 仅保留低4位（行选信号只占低4位，高4位无效）
                            ; 可转换：改0FH为07H→仅读取低3位（适配3行键盘）
    CMP AL,0FH              ; 判断是否有按键：0FH=无按键（4行都为1），非0FH=有按键
                            ; 可转换：改0FH为07H→适配3行键盘的无按键判断
    JE M2                   ; 无按键则跳转到M2（跳过按键处理）
    CALL JUDGE              ; 有按键则调用JUDGE子程序（解析按键并执行功能）
    JMP M3                  ; 跳转到M3（跳过无按键的互斥处理）
M2:                         ; 无按键处理标号
    CMP FLAG,00H            ; 判断按键互斥标志（0=无互斥，非0=需递减）
    JE M3                   ; 无互斥则跳转到M3
    DEC FLAG                ; 互斥标志递减（FLAG=MAX(0,FLAG-1)，防止重复触发）
                            ; 可转换：改DEC FLAG为SUB FLAG,2→互斥次数减少2次，加快按键响应
M3:                         ; 显示处理标号
    CALL SHOW               ; 调用SHOW子程序（刷新6位数码管显示）
    POP CX                  ; 出栈恢复扫描次数
    POP AX                  ; 出栈恢复列选信号
    CALL DELAY              ; 调用短延时子程序（防止扫描过快导致的按键误判）
    LOOP M1                 ; CX减1，非0则跳回M1（继续扫描下一列）
    JMP MAIN                ; 跳回MAIN（无限循环，持续扫描+显示）

; ===================== 区块8：JUDGE子程序块【按键解析的核心，改这里扩展/修改按键功能】 =====================
; 功能：将键盘扫描的行选/列选信号转换为0~F的按键索引，调用对应功能子程序
JUDGE:
    NOT AL                  ; 对行选信号取反（无按键=0FH→00H，有按键=非0FH→对应行号）
    AND AX,0FH              ; 仅保留低4位（清除高12位干扰）
    ; 按键行号判断：将行选信号转换为行索引（01H=行1，02H=行2，04H=行3，08H=行4）
    CMP AL,01H
    JE D1
    CMP AL,02H
    JE D2
    CMP AL,04H
    JE D3
    CMP AL,08H
    JE D4
D1:
    MOV AL,04H              ; 行1对应列偏移04H
                            ; 可转换：改04H为00H→行1对应列偏移00H（适配不同键盘接线）
    JMP D5
D2:
    MOV AL,08H              ; 行2对应列偏移08H
    JMP D5
D3:
    MOV AL,0CH              ; 行3对应列偏移0CH
    JMP D5
D4:
    MOV AL,10H              ; 行4对应列偏移10H
    JMP D5
D5:
    ADD AL,CL               ; 列偏移+列数（CL=CX，当前扫描列）→按键的全局索引
    SUB AL,05H              ; 索引修正（减去偏移，得到0~F的连续索引）
    MOV SI,AX               ; SI保存按键的段码表索引（0=0，1=1…15=F）
                            ; 可转换：改SUB AL,05H为SUB AL,04H→索引修正值适配不同键盘接线
    CMP FLAG,00H            ; 判断按键互斥标志（0=无互斥，非0=重复触发）
    JNE JUDGE_END           ; 有互斥则跳转到结束（跳过按键处理）
    
    ; 按键功能分支：0~9键输初值，A~F键执行对应功能（实验核心的按键逻辑）
    CMP SI,0AH              ; 判断是否为A键（0AH）
    JNE D0A
    CALL PRESS_A            ; 调用A键子程序（启动/取消计时）
    JMP JUDGE_END
D0A:
    CMP SI,0BH              ; 判断是否为B键（0BH）
    JNE D0B
    CALL PRESS_B            ; 调用B键子程序（暂停/继续计时）
    JMP JUDGE_END
D0B:
    CMP SI,0CH              ; 判断是否为C键（0CH）
    JNE D0C
    CALL PRESS_C            ; 调用C键子程序（退出程序）
    JMP JUDGE_END
D0C:
    CMP SI,0DH              ; 判断是否为D键（0DH，实验未定义功能）
    JNE D0D
    CALL PRESS_D            ; 调用D键子程序（预留功能）
    JMP JUDGE_END
D0D:
    CMP SI,0EH              ; 判断是否为E键（0EH，实验未定义功能）
    JNE D0E
    CALL PRESS_E            ; 调用E键子程序（预留功能）
    JMP JUDGE_END
D0E:
    CMP SI,0FH              ; 判断是否为F键（0FH，实验未定义功能）
    JNE D0F
    CALL PRESS_F            ; 调用F键子程序（预留功能）
    JMP JUDGE_END
D0F:
    ; 0~9键处理：仅初始状态（STATE=00H）可输入计时初值
    CMP STATE,00H
    JNE JUDGE_END           ; 非初始状态则跳过（禁止输初值）
    
    LEA BX,TAB              ; BX指向段码表（无实际作用，原代码预留）
    ; 计时初值输入逻辑：T2（分个位）→T1（分十位），实现1~99分钟输入
    MOV AL,T2
    MOV T1,AL               ; T2的值移到T1（分十位）
    MOV AX,SI
    MOV T2,AL               ; 新按键值写入T2（分个位）
                            ; 可转换：互换T1/T2→分十位/个位输入顺序对调；改T2为T3→初值输入到秒位
JUDGE_END:
    MOV FLAG,04H            ; 设置互斥标志初值：4次扫描内不重复触发（防抖动）
                            ; 可转换：改04H为02H→互斥次数为2，加快按键响应；改08H→更防抖动
    RET                     ; 子程序返回

; ===================== 区块9：A键子程序块【启动/取消计时，实验核心功能1】 =====================
; 功能：初始状态按A键启动计时，计时/暂停中按A键回初始状态
PRESS_A:
    CMP STATE,02H           ; 判断是否为暂停状态（02H）
    JE D_4                  ; 是则跳转到D_4（回初始）
    CMP STATE,01H           ; 判断是否为计时状态（01H）
    JE D_4                  ; 是则跳转到D_4（回初始）
    CMP STATE,00H           ; 判断是否为初始状态（00H）
    JNE D_END               ; 非初始状态则结束
    
    ; 初值有效性判断：防止输入00分钟
    CMP T2,10H              ; T2（分个位）是否为无效值（10H）
    JE D_END                ; 是则结束（未输入初值）
    CMP T2,00H              ; T2是否为0
    JE D_1                  ; 是则跳转到D_1（进一步判断T1）
    CMP T1,10H              ; T1（分十位）是否为无效值
    JE D_2                  ; 是则跳转到D_2（T1置0）
    CMP T1,00H
    JE D_3
D_1:
    CMP T1,10H              ; T1是否为无效值
    JE D_END                ; 是则结束（未输入初值）
    CMP T1,00H              ; T1是否为0
    JE D_END                ; 是则结束（初值为00，禁止启动）
    JMP D_3                 ; 否则跳转到D_3（启动计时）
D_2:
    MOV T1,00H              ; T1置0（如输入5→05分钟）
    JMP D_3                 ; 启动计时
D_3:
    MOV STATE,01H           ; 设置状态为计时中（01H）
    MOV T3,00H              ; 秒的十位初始化为0
    MOV T4,00H              ; 秒的个位初始化为0
                            ; 可转换：改T3为05H→初始秒为50秒；改T4为30H→初始秒为30秒
    JMP D_END
D_4:
    MOV STATE,00H           ; 恢复为初始状态（00H）
    MOV T1,10H              ; 分十位置为无效值（熄灭）
    MOV T2,10H              ; 分个位置为无效值（熄灭）
    MOV T3,10H              ; 秒十位置为无效值（熄灭）
    MOV T4,10H              ; 秒个位置为无效值（熄灭）
    JMP D_END
D_END:
    RET                     ; 子程序返回

; ===================== 区块10：B键子程序块【暂停/继续计时，实验核心功能2】 =====================
; 功能：计时中按B键暂停，暂停中按B键继续
PRESS_B:
    CMP STATE,01H           ; 判断是否为计时状态（01H）
    JE E1                   ; 是则跳转到E1（暂停）
    CMP STATE,02H           ; 判断是否为暂停状态（02H）
    JE E2                   ; 是则跳转到E2（继续）
    JMP E3                  ; 非计时/暂停状态则结束
E1:
    MOV STATE,02H           ; 设置状态为暂停中（02H）
    JMP E3
E2:
    MOV STATE,01H           ; 恢复为计时中（01H）
    JMP E3
E3:
    RET                     ; 子程序返回

; ===================== 区块11：C键子程序块【退出程序，实验核心功能3】 =====================
; 功能：任何时刻按C键熄灭数码管并退出程序

    MOV T2,10H              ; 分个位置为无效值
    MOV T3,10H              ; 秒十位置为无效值
    MOV T4,10H              ; 秒个位置为无效值
    JMP EXIT                ; 跳转到退出程序

; ===================== 区块14：CLEAR_SHOW子程序块【熄灭数码管，实验辅助功能】 =====================
; 功能：输出熄灭码到所有数码管，实现“无显示”效果
CLEAR_SHOW:
    MOV AL,11011111B        ; 数码管位选初始值：选择最右端T1
                            ; 可转换：改11011111B为10111111B→从最左端开始熄灭
    MOV CX,06H              ; 数码管熄灭次数：6次（对应6位显示）
                            ; 可转换：改06H为04H→只熄灭4位（保留T5/T6）
CS2:
    MOV DX,A8255            ; DX指向8255的A口（输出位选信号）
    OUT DX,AL               ; 输出位选信号（选中当前数码管）
    SHR AL,1                ; 位选信号右移一位（选中下一个数码管）
    OR AL,11000000B         ; 保留高2位为1，防止干扰
    PUSH AX                 ; 压栈保存位选信号
    MOV DX,B8255            ; DX指向8255的B口（输出段选信号）
    MOV AL,00H              ; 数码管熄灭码（00H）
    OUT DX,AL               ; 输出熄灭码（熄灭当前数码管）
    POP AX                  ; 出栈恢复位选信号
    CALL DELAY              ; 短延时（确保熄灭效果）
    LOOP CS2                ; CX减1，非0则跳回CS2
    RET                     ; 子程序返回

; ===================== 区块15：SHOW子程序块【数码管刷新显示的核心，改这里控制显示格式】 =====================
; 功能：根据TAB2的缓存值，将0~F的索引转换为段码，输出到6位数码管
SHOW:
    MOV AL,11011111B        ; 数码管位选初始值：选择最右端T1
                            ; 可转换：改11011111B为10111111B→从最左端开始显示（反向）
    MOV SI,00H              ; 数码管缓存索引（0=T1，1=T2…5=T6）
    MOV CX,06H              ; 数码管显示次数：6次（对应6位显示）
                            ; 可转换：改06H为04H→只显示4位（隐藏T5/T6）；改08H→扩展为8位显示
S2:
    MOV DX,A8255            ; DX指向8255的A口（输出位选信号）
    OUT DX,AL               ; 输出位选信号（选中当前数码管）
    SHR AL,1                ; 位选信号右移一位（选中下一个数码管）
    OR AL,11000000B         ; 保留高2位为1，防止干扰
    PUSH AX                 ; 压栈保存位选信号
    
    ; 段码转换逻辑：TAB2的缓存值（0~F）→TAB的段码值
    LEA BX,TAB2             ; BX指向数码管缓存TAB2的首地址
    MOV AX,00H
    MOV AL,[BX+SI]          ; 读取当前索引的缓存值（T1~T6）
    PUSH SI                 ; 压栈保存索引
    LEA BX,TAB              ; BX指向段码表TAB的首地址
    MOV SI,AX               ; SI=缓存值（0~F）
    MOV AL,[BX+SI]          ; 读取对应的段码值
    MOV DX,B8255            ; DX指向8255的B口（输出段码信号）
    OUT DX,AL               ; 输出段码值（当前数码管显示对应数字）
    POP SI                  ; 出栈恢复索引
    POP AX                  ; 出栈恢复位选信号
    
    INC SI                  ; 索引+1（读取下一个缓存值）
    CALL DELAY              ; 短延时（防止显示闪烁）
    LOOP S2                 ; CX减1，非0则跳回S2
    RET                     ; 子程序返回

; ===================== 区块16：MIR6中断服务程序块【1秒计时中断的核心，实验核心功能5】 =====================
; 功能：每秒触发一次，调用TIME_DEC实现倒计时，仅计时中（01H）执行
MIR6:
    STI                     ; 开中断（允许嵌套，固定操作）
    PUSH CX                 ; 压栈保存CX（防止覆盖主程序）
    PUSH BX                 ; 压栈保存BX（防止覆盖主程序）
    PUSH AX                 ; 压栈保存AX（防止覆盖主程序）
    
    CMP STATE,01H           ; 判断是否为计时状态（01H）
    JNE MIR_END             ; 非计时状态则结束
    
    CALL TIME_DEC           ; 调用计时减法子程序（实现1秒倒计时）
    JMP MIR_END
MIR_END:
    POP AX                  ; 出栈恢复AX
    POP BX                  ; 出栈恢复BX
    POP CX                  ; 出栈恢复CX
    IRET                    ; 中断返回（区别于普通RET，固定操作）

; ===================== 区块17：TIME_DEC子程序块【倒计时减法逻辑，实验核心功能6】 =====================
; 功能：实现“秒个位→秒十位→分个位→分十位”的借位减法，倒计时至0000时调用TIME_END
TIME_DEC:
    CMP T4,00H              ; 判断秒个位（T4）是否为0
    JNE DEC_T4              ; 非0则直接减1
    MOV T4,09H              ; 秒个位置9（借位）
    CMP T3,00H              ; 判断秒十位（T3）是否为0
    JNE DEC_T3              ; 非0则直接减1
    MOV T3,05H              ; 秒十位置5（借位，60秒=1分钟）
    CMP T2,00H              ; 判断分个位（T2）是否为0
    JNE DEC_T2              ; 非0则直接减1
    MOV T2,09H              ; 分个位置9（借位）
    CMP T1,00H              ; 判断分十位（T1）是否为0
    JNE DEC_T1              ; 非0则直接减1
    CALL TIME_END           ; 借无可借（0000），调用计时结束子程序
    JMP DEC_END
; 各数位减法逻辑（无借位）
DEC_T4:
    DEC T4                  ; 秒个位减1
    JMP DEC_END
DEC_T3:
    DEC T3                  ; 秒十位减1
    JMP DEC_END
DEC_T2:
    DEC T2                  ; 分个位减1
    JMP DEC_END
DEC_T1:
    DEC T1                  ; 分十位减1
DEC_END:
    RET                     ; 子程序返回

; ===================== 区块18：TIME_END子程序块【计时结束处理，实验核心功能7】 =====================
; 功能：倒计时至0000后闪烁3次，回到初始状态
TIME_END:
    PUSH CX                 ; 压栈保存CX（防止覆盖）
    MOV STATE,03H           ; 设置状态为计时结束（03H）
    MOV T4,10H              ; 秒个位置为无效值（熄灭）
    MOV T3,10H              ; 秒十位置为无效值（熄灭）
    MOV T2,10H              ; 分个位置为无效值（熄灭）
    MOV T1,10H              ; 分十位置为无效值（熄灭）
    
    MOV CX,03H              ; 闪烁次数：3次（实验要求）
                            ; 可转换：改03H为05H→闪烁5次；改01H→闪烁1次
TIME_SHIN:
    CALL TIME_OUT0          ; 调用显示0000子程序
    CALL DELAY_LONG         ; 长延时（闪烁亮的时间）
    CALL TIME_CLEAR         ; 调用清除显示子程序
    CALL DELAY_LONG         ; 长延时（闪烁灭的时间）
    LOOP TIME_SHIN          ; CX减1，非0则跳回TIME_SHIN
    
    MOV STATE,00H           ; 恢复为初始状态（00H）
    POP CX                  ; 出栈恢复CX
    RET                     ; 子程序返回

; ===================== 区块19：TIME_OUT0子程序块【显示0000，实验辅助功能】 =====================
; 功能：输出段码3FH（0）到数码管1-4，实现0000显示
TIME_OUT0:
    PUSH AX                 ; 压栈保存AX
    MOV AL,11110000B        ; 位选信号：选中数码管1-4
    MOV DX,A8255            ; DX指向8255的A口
    OUT DX,AL               ; 输出位选信号
    MOV AL,3FH              ; 段码：0（3FH）
    MOV DX,B8255            ; DX指向8255的B口
    OUT DX,AL               ; 输出段码信号
    POP AX                  ; 出栈恢复AX
    RET                     ; 子程序返回

; ===================== 区块20：TIME_CLEAR子程序块【清除0000显示，实验辅助功能】 =====================
; 功能：输出熄灭码00H到数码管1-4，实现灭屏
TIME_CLEAR:
    PUSH AX                 ; 压栈保存AX
    MOV AL,11110000B        ; 位选信号：选中数码管1-4
    MOV DX,A8255            ; DX指向8255的A口

## 需求2：把**2位0-99倒计时**改成**正计时**（粘代码+改1行调用）
还是用“复制倒计时改加法”的思路，**新增一个正计时子程序`TIME_ADD`**，再改中断里的调用名，原有倒计时代码保留（不想用了再换回来）。

### 步骤1：新增「2位0-99正计时子程序`TIME_ADD`」

```asm

TIME_ADD:
    CMP T2,09H              ; 判断分个位T2是否到9
    JNE INC_T2              ; 没到9，直接加1
    MOV T2,00H              ; 到9了，分个位归0（进位）

        JMP ADD_END
    INC_T1:

    ADD_END:
        RET                 ; 子程序返回
```

; ====================== 区块7：显示/清屏区 ======================

	MOV AL,11011111B	;数码管位选码：初始选第1个数码管 【可修改】改11011111B为10111111B，从第6个数码管开始清屏

; ====================== 区块9：计时逻辑区 ======================
; 功能：实现倒计时、计时结束闪烁，是考试修改计时规则的核心区

; 子区块9.1：倒计时程序（TIME_DEC）
TIME_DEC:
	; 借位减法：秒个位→秒十位→分个位→分十位
	CMP T4,00H			;判断秒个位是否为0 【可修改】改T4为T2，先减分个位（跳过秒）；改00H为05H，秒个位到5就借位
		JNE DEC_T4
		MOV T4,09H		;秒个位归9 【可修改】改09H为05H，秒个位最大为5（0-5）
	CMP T3,00H			;判断秒十位是否为0 【可修改】改T3为T1，先减分十位；改00H为03H，秒十位到3就借位
		JNE DEC_T3
		MOV T3,05H		;秒十位归5 【可修改】改05H为09H，秒十位最大为9（0-9，变成0-99秒）
	CMP T2,00H			;判断分个位是否为0 【可修改】改00H为03H，分个位到3就借位
		JNE DEC_T2
		MOV T2,09H		;分个位归9 【可修改】改09H为05H，分个位最大为5（0-5）
	CMP T1,00H			;判断分十位是否为0 【可修改】改00H为01H，分十位到1就借位
		JNE DEC_T1
		
	CALL TIME_END		;借无可借，调用计时结束程序 【可修改】改TIME_END为TIME_LOOP，计时结束后循环
	JMP DEC_END			;跳转到结束 【不可改】固定写法
		
	DEC_T4:
		DEC T4			;秒个位减1 【可修改】改DEC为INC，秒个位加1（正计时）
		JMP DEC_END
	DEC_T3:
		DEC T3			;秒十位减1 【可修改】改DEC为INC，秒十位加1（正计时）
		JMP DEC_END
	DEC_T2:
		DEC T2			;分个位减1 【可修改】改DEC为INC，分个位加1（正计时）
		JMP DEC_END
	DEC_T1:
		DEC T1			;分十位减1 【可修改】改DEC为INC，分十位加1（正计时）
		
	DEC_END:
		RET				;返回 【不可改】固定写法

; 子区块9.2：计时结束程序（TIME_END）
TIME_END:
	PUSH CX				;保存寄存器 【不可改】固定写法
	
	MOV STATE,03H		;设置为结束状态 【可修改】改03H为04H，设置为循环计时状态
	MOV T4,10H			;秒个位熄灭 【可修改】改10H为00H，结束状态显示0
	MOV T3,10H			;秒十位熄灭 【可修改】改10H为00H，结束状态显示0
	MOV T2,10H			;分个位熄灭 【可修改】改10H为00H，结束状态显示0
	MOV T1,10H			;分十位熄灭 【可修改】改10H为00H，结束状态显示0
	
		MOV CX, 03H		;闪烁3次 【可修改】改03H为05H，闪烁5次；改03H为01H，闪烁1次；改03H为00H，不闪烁
	TIME_SHIN:
		CALL TIME_OUT0	;显示0000 【可修改】改TIME_OUT0为TIME_OUT1，显示其他数字（如1111）
		CALL DELAY_LONG	;长延时 【可修改】改DELAY_LONG为DELAY，闪烁间隔变短（更快）
		CALL TIME_CLEAR	;清屏 【可修改】改TIME_CLEAR为TIME_OUT0，闪烁变为0000和0000交替（无效果）
		CALL DELAY_LONG	;长延时 【可修改】改DELAY_LONG为DELAY，闪烁间隔变短
		LOOP TIME_SHIN	;循环闪烁 【不可改】固定写法
		
	MOV STATE,00H		;回到初始状态 【可修改】改00H为01H，结束后直接重新开始计时（循环）
	POP CX				;还原寄存器 【不可改】固定写法
	RET					;返回 【不可改】固定写法
