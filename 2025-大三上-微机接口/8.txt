A8255 EQU 0600H
B8255 EQU 0602H
C8255 EQU 0604H  
MODE8255 EQU 0606H
 
DATA SEGMENT
    TAB:       ;段码表
        DB 3FH    ;0
        DB 06H    ;1
        DB 5BH    ;2
        DB 4FH    ;3
        DB 66H    ;4
        DB 6DH    ;5
        DB 7DH    ;6
        DB 07H    ;7
        DB 7FH    ;8
        DB 6FH    ;9
DATA ENDS
 
CODE SEGMENT
    ASSUME CS:CODE,DS:DATA
START:
    MOV AX,DATA
    MOV DS,AX   
        
    MOV DX,MODE8255
    MOV AL,81H
    OUT DX,AL    
    
    MOV DX,A8255
    MOV AL,00H
    OUT DX,AL
    
    MOV DX,B8255
    MOV AL,00H
    OUT DX,AL
 
    LEA BX,TAB
    MOV SI,00H
    MOV AL,11011111B  ;初始选择最高位数码管，即最右端的数码管    
 
MAIN:  
    CMP AL,01111111B  ;判断数码管是否到达最左端
    JE AA0
    
    MOV DX,A8255 
    OUT DX,AL         ;选择数码管
    
    PUSH AX
    MOV DX,B8255
    MOV AL,[BX+SI]
    OUT DX,AL
    CALL DELAY  
    POP AX 
     
    ROR AL,01H
    JMP MAIN
    
AA0:
    INC SI
    MOV AL,11011111B   
    CMP SI,0AH
	JE AA1
	JMP MAIN
AA1:
    MOV SI,00H  
    JMP MAIN     
 
DELAY:
    PUSH CX
    MOV CX,0FFFFH
D0:
    PUSH AX
    POP AX
    LOOP D0
    POP CX
    RET
    
CODE ENDS
     END START

; ====================== 区块1：硬件地址定义区【基础硬件配置，实验箱适配核心】 ======================
; 作用：给8255的A/B/C口和方式控制口分配固定操作地址，是程序控制数码管/开关的“门牌号”，需与实验箱实际接线一致
; 【每一行可灵活修改的方式+具体效果（实验箱适配/功能调整）】：
; 1. A8255 EQU 0600H → 修改0600H为其他十六进制数（如0640H/0200H/0300H）：
;    - 0640H：适配实验箱8255 A口接数码管位选的接线（本程序中A口输出位选，核心！）
;    - 0200H：适配老式实验箱的8255地址，仅需改此值，其他代码无需动
;    - 核心作用：更换8255 A口的硬件地址，保证程序能控制数码管的位选信号（X1~X6）
; 2. B8255 EQU 0602H → 修改0602H为其他值（如0642H/0202H）：
;    - 本程序中B口输出数码管的段码（A~G+Dp），若实验箱段码接A口、位选接B口，需互换A8255/B8255的定义，并同步修改后续的DX输出地址
; 3. C8255 EQU 0604H → 修改0604H为其他值（如0644H/0204H）：
;    - 本程序中C口读取开关状态（6个开关接C口），若开关接A口/B口，需改此值并同步修改8255的方式控制字
; 4. MODE8255 EQU 0606H → 修改0606H为其他值（如0646H/0206H）：
;    - 8255的方式控制字写入地址，是并行口的“配置入口”，必须与实验箱的8255控制口一致，改此值后需同步修改初始化的DX值
A8255 EQU 0600H   ; 8255 A口：数码管位选输出口（X1~X6），改此值适配实验箱接线
B8255 EQU 0602H   ; 8255 B口：数码管段码输出口（A~G+Dp），改此值适配实验箱接线
C8255 EQU 0604H   ; 8255 C口：开关状态输入口，改此值适配实验箱接线
MODE8255 EQU 0606H; 8255方式控制口，改此值适配实验箱接线

; ====================== 区块2：数据段定义区【数码管显示核心数据，考试改分关键！】 ======================
; 作用：存储**学号后六位的共阴极段码表**和**开关状态标志位**，是数码管显示内容的“数据源”，也是考试中最常修改的区域
; 【每一行可灵活修改的方式+具体效果（覆盖99%的考题变种）】：
; 一、段码表TAB的修改（核心！考题必改：显示学号/字母/加小数点/熄灭）
; 1. 段码值修改：原代码是6个06H（对应数字1），需改为**自己的学号后六位的共阴极段码**，段码对照表（题目已给）：
;    - 0：3FH  1：06H  2：5BH  3：4FH  4：66H  5：6DH  6：7DH  7：07H  8：7FH  9：6FH
;    - A：77H  B：7CH  C：39H  D：5EH  E：79H  F：71H  熄灭：00H  小数点：段码|80H（如0带小数点：3FH+80H=BFH）
;    - 举例1：学号后六位为202468 → 段码依次为5BH、3FH、5BH、66H、7DH、7FH，需把TAB下的6行改为对应值
;    - 举例2：显示字母A-F → 段码依次为77H、7CH、39H、5EH、79H、71H
;    - 举例3：第3位加小数点（如数字4带点）→ 段码改为66H+80H=E6H
;    - 举例4：第5位熄灭 → 段码改为00H
; 2. 段码数量修改：若考题要求4位/8位数码管，需增删TAB下的段码行（如4位则保留4行，8位则新增2行）
; 二、FLAG标志位的修改
; 1. FLAG DB 00H → 变量名可改（如SWITCH_FLAG DB 00H），仅提升可读性，无功能影响
; 2. 可新增变量：如DISP_CNT DB 06H（存储数码管数量，适配4/8位数码管）、DELAY_VAL DW 02FFH（存储延时值，统一控制刷新率）
DATA SEGMENT
   TAB:                    ; 学号后六位的共阴极段码表（考题必改！核心！）
    DB 06H	; 第1位：数字1（改此值为自己学号的段码，如2→5BH）
    DB 06H	; 第2位：数字1（改此值为自己学号的段码）
    DB 06H	; 第3位：数字1（改此值为自己学号的段码）
    DB 06H	; 第4位：数字1（改此值为自己学号的段码）
    DB 06H	; 第5位：数字1（改此值为自己学号的段码）
    DB 06H	; 第6位：数字1（改此值为自己学号的段码）
   FLAG DB 00H; 存储C口读取的6个开关状态（变量名可改，无功能影响）
DATA ENDS   

; ====================== 区块3：代码段声明区【汇编程序的固定格式，不可乱改】 ======================
; 作用：告诉编译器代码段的名称、寄存器的归属关系（CS对应代码段，DS对应数据段），是汇编程序的“骨架”
; 【可灵活修改的行及效果（仅可读性调整，无功能影响）】：
; 1. CODE SEGMENT → 段名可改（如MAIN_CODE SEGMENT），仅提升可读性
; 2. ASSUME CS:CODE,DS:DATA → 若数据段名改了（如DATA_SEG），需同步改DS:DATA_SEG，否则编译器报错
; 3. CODE ENDS → 需与段名同步修改（如MAIN_CODE ENDS），否则汇编报错
CODE SEGMENT
    ASSUME CS:CODE,DS:DATA  ; 绑定CS到代码段，DS到数据段（数据段名改则同步改）

; ====================== 区块4：数据段初始化区【绑定数据源，让程序能访问段码表】 ======================
; 作用：将数据段的段地址加载到DS寄存器，让程序能读取TAB段码表和FLAG变量，是“程序访问数据的桥梁”
; 【可灵活修改的行及效果（无功能影响，仅格式调整）】：
; 1. MOV AX,DATA → 若数据段名改了（如DATA_SEG），需同步改AX,DATA_SEG
; 2. MOV DS,AX → 固定指令，不可修改，若删除则程序无法访问数据段的TAB和FLAG
START:
 	MOV AX,DATA             ; 加载数据段段地址到AX（数据段名改则同步改）
	MOV DS,AX               ; DS绑定到数据段（固定指令，不可删）

; ====================== 区块5：8255初始化区【数码管/开关的输入输出模式配置，核心！】 ======================
; 作用：配置8255的工作模式，指定**A口输出（位选）、B口输出（段码）、C口输入（开关）**，是程序能控制数码管/读取开关的前提
; 【每一行可灵活修改的方式+具体效果（适配不同硬件接线/输入输出需求）】：
; 1. MOV DX,MODE8255 → 若MODE8255地址改了（如0646H），需同步改此DX值，否则8255初始化失败
; 2. MOV AL,89H → 修改89H为其他8255方式控制字，**彻底改变输入输出模式**（核心！需匹配硬件接线）：
;    8255方式控制字格式（8位）：D7 D6 D5 D4 D3 D2 D1 D0
;    - D7=1：方式控制字标志（固定）
;    - D6D5=00：A口工作在方式0
;    - D4=1：A口输出（位选）；D4=0：A口输入
;    - D3=0：A口的PC7~PC4输出（本程序未用）
;    - D2=00：B口工作在方式0
;    - D1=1：B口输出（段码）；D1=0：B口输入
;    - D0=1：B口的PC3~PC0输入（开关接C口低4位）；D0=0：C口低4位输出
;    原代码89H=10001001B → A口输出、B口输出、C口输入（适配6个开关接C口）
;    变种修改举例：
;    - 若开关接C口高4位 → 控制字改为8DH=10001101B
;    - 若数码管位选接B口、段码接A口 → 控制字改为99H=10011001B（A口输入、B口输出，需同步互换A8255/B8255的定义）
;    - 若需静态显示（仅1位数码管）→ 控制字不变，仅修改后续位选逻辑
; 3. OUT DX,AL → 固定指令，将控制字写入8255控制口，不可修改，若删除则8255初始化失败
	MOV DX,MODE8255         ; 指向8255方式控制口（MODE8255地址改则同步改）
	MOV AL,89H	            ; 8255控制字：A/B口输出（位选/段码），C口输入（开关），改此值换输入输出模式
	OUT DX,AL               ; 写入控制字，完成8255初始化（固定指令，不可删）

; ====================== 区块6：段码表地址绑定区【指向显示数据源，让程序能读取学号段码】 ======================
; 作用：将TAB段码表的偏移地址加载到BX寄存器，让程序能通过[BX+SI]读取第SI位的段码，是“程序访问学号段码的指针”
; 【可灵活修改的行及效果（适配段码表位置调整）】：
; 1. LEA BX,TAB → 若段码表名改了（如STU_NUM），需同步改LEA BX,STU_NUM
;    也可改为MOV BX,OFFSET TAB，功能完全相同，仅指令格式不同
	LEA BX,TAB              ; BX指向段码表TAB的首地址（段码表名改则同步改）

; ====================== 区块7：主循环初始化区【数码管动态扫描的参数初始化，核心！】 ======================
; 作用：初始化**位选码、段码偏移量、扫描次数**，是6位数码管动态扫描的“初始参数”，决定了扫描的起始位和次数
; 【每一行可灵活修改的方式+具体效果（改扫描顺序/数码管数量，考题高频变种）】：
; 1. MOV AL,11011111B → 修改此位选码，**改变数码管的扫描起始位**（核心！）：
;    位选码说明（6位数码管X1~X6对应AL的D5~D0位，共阴极数码管：位选=0点亮，=1熄灭）：
;    - 11011111B（0DFH）：第1位（X1）位选=0，其余=1 → 从第1位开始扫描
;    - 10111111B（0BFH）：从第2位开始扫描
;    - 01111111B（07FH）：从第6位开始扫描
;    - 变种：反向扫描（从第6位到第1位）→ 初始位选码改为01111111B，移位指令改为ROL AL,01H
; 2. MOV SI,00H → SI是段码表的偏移量，初始为0表示从第1位段码开始读取，若改为01H则从第2位开始显示
; 3. MOV CX,06H → 修改06H为其他值，**改变数码管的扫描数量**（考题高频！）：
;    - 04H：4位数码管显示（需同步删TAB段码表的2行）
;    - 08H：8位数码管显示（需同步增TAB段码表的2行）
;    - 03H：3位数码管显示（需同步删TAB段码表的3行）
MAIN:	
	MOV AL,11011111B        ; 位选码初始值：第1位数码管位选=0（改此值换扫描起始位）
	MOV SI,00H              ; 段码表偏移量初始值：从第1位段码开始读取
	MOV CX,06H              ; 扫描次数：6位数码管（改06H为04H/08H换数码管数量，核心！）

; ====================== 区块8：开关状态读取区【获取开关信号，决定数码管亮灭，核心逻辑！】 ======================
; 作用：从8255 C口读取6个开关的状态，存入FLAG变量，是“开关控制数码管亮灭的数据源”
; 【每一行可灵活修改的方式+具体效果（适配开关接线/读取逻辑）】：
; 1. PUSH DX/PUSH AX → 保护DX/AX寄存器，防止读取开关时覆盖原有值，可删除但会导致位选码/扫描参数丢失
; 2. MOV DX,C8255 → 若C8255地址改了（如0644H），需同步改此DX值，否则无法读取开关状态
; 3. IN AL,DX → 固定指令，从C口读取开关状态，若改为OUT DX,AL则程序崩溃（C口是输入）
; 4. MOV FLAG,AL → 将开关状态存入FLAG，若删除则后续无法使用开关信号，需直接用AL
; 5. POP AX/POP DX → 恢复DX/AX寄存器，与前面的PUSH对应，删除则寄存器值错误
AA0:
	PUSH DX                 ; 保护DX寄存器（位选码的DX）
	PUSH AX                 ; 保护AX寄存器（位选码的AL）
	MOV DX,C8255            ; 指向8255 C口（开关输入口，地址改则同步改）
	IN AL,DX                ; 读取6个开关的状态到AL（固定指令，不可改）
	MOV FLAG,AL             ; 将开关状态存入FLAG变量（可删除，后续直接用AL）
	POP AX                  ; 恢复AX寄存器（位选码）
	POP DX                  ; 恢复DX寄存器（位选输出口）

; ====================== 区块9：开关与位码匹配区【判断是否点亮当前数码管，考试变种核心！】 ======================
; 作用：将**当前位选码**与**开关状态**做逻辑运算，判断“对应开关是否为1”，从而决定是否点亮当前数码管，是开关控制的核心逻辑
; 【每一行可灵活修改的方式+具体效果（改亮灭条件/匹配逻辑，考题高频变种）】：

; 2. MOV BL,[FLAG] → 将开关状态加载到BL，若删除则需直接用AL（需保证AL未被修改）

;    - 仅当开关为1且位选为1时点亮 → 删除NOT BH，直接TEST BH,BL

    MOV DX, MODE8254	        ; 指向8254方式控制口（硬件固定则无需改）
    MOV AL, 36H               ; 8254控制字：计数器0、方式3（改此值换工作方式，核心！）
    OUT DX, AL                ; 写入8254控制字（固定指令，不可改）
    MOV DX, A8254 	        ; 指向8254计数器0（硬件固定则无需改）
    MOV AL, 00H               ; 定时初值低8位（配合高8位，无需单独改）
    OUT DX, AL                ; 写入低8位（固定指令，不可改）
    MOV AL, 48H               ; 定时初值高8位（改此值调定时时间，核心！如2秒→90H）

; ====================== 区块7：数据段绑定+8255初始化区【数码管驱动核心，显示变种改此部分！】 ======================
; 作用：绑定数据段（让程序能访问TABLE1段码表），配置8255的工作模式（A/B口输出），是数码管显示的前提
; 【每一行可灵活修改的方式+效果（8255模式调整，考试低频变种）】：
; 1. MOV AX,DATA/MOV DS,AX → 绑定数据段，不可删除，否则程序无法访问TABLE1
; 2. LEA SI,TABLE1 → SI指向段码表首地址，若段码表名改了（如SEG_TABLE），需同步改LEA SI,SEG_TABLE
; 3. MOV DX,MODE8255/MOV AL,81H/OUT DX,AL → 81H=8255方式控制字，**改此值可换输入输出模式**：
;    - 81H=10000001B → A口输出（位选）、B口输出（段码）、C口输入（备用），本程序用此
;    - 80H=10000000B → A/B/C口均输出，无功能影响，仅扩展C口为输出
;    - 89H=10001001B → A/B口输出、C口输入，与81H功能一致
; 4. MOV POS,11111110B → 初始化位选码（X1数码管点亮，11111110B=0FEH），**改此值可换“123”的起始显示位置**：
;    - 改11111101B（0FDH）→ 从X2数码管开始显示“123”

    MOV AX,DATA		            ; 加载数据段地址（固定指令，不可改）
    MOV DS,AX                 ; DS绑定到数据段（固定指令，不可改）
 
    LEA SI,TABLE1             ; SI指向段码表TABLE1首地址（段码表名改则同步改）
    
    ;8255初始化
    MOV DX,MODE8255           ; 指向8255方式控制口（硬件固定则无需改）
    MOV AL,81H                ; 8255控制字：A/B口输出（位选/段码），C口输入（改80H则C口输出）
    OUT DX,AL                 ; 写入8255控制字（固定指令，不可改）
    
    MOV POS,11111110B	        ; 初始化位选码：X1数码管点亮（改此值换起始显示位置，核心！）

; ====================== 区块8：主循环扫描显示区【数码管动态扫描核心，显示位数变种改此部分！】 ======================
; 作用：循环读取段码表TABLE1，调用DISP子程序实现“123”的动态扫描显示，是数码管能同时亮的关键
; 【每一行可灵活修改的方式+效果（显示位数/扫描逻辑调整，考试高频变种！）】：
; 1. MOV AL,[POS] → 读取当前位选码，不可修改
; 2. MOV BX, 00H → BX是段码表的偏移量，初始为0（从第一个段码开始读），改01H则从第二个段码开始显示
; 3. MOV CX,03H → 扫描次数（对应显示3位数字），**改此值可换显示位数，核心！**：
;    - 改CX=02H → 显示2位数字（如“12”），需同步删除TABLE1第三行
;    - 改CX=04H → 显示4位数字（如“1234”），需同步新增TABLE1第四行
; 4. MOV SITE,AL → 临时存储位选码到SITE，不可修改
; 5. PUSH AX/MOV AL,[BX+SI]/MOV NUM,AL/POP AX → 读取段码表的第BX位段码，存入NUM，不可修改
; 6. CALL DISP → 调用显示子程序，不可修改
; 7. INC BX → 段码表偏移量+1，读取下一个段码，不可修改
; 8. ROL AL,1 → 循环左移位选码（控制“123”右移的基础），**改此指令可换扫描方向，核心！**：
;    - 改ROR AL,1 → 循环右移位选码（扫描方向反转）
;    - 改SHL AL,1 → 逻辑左移（仅扫描前几位，无循环）
; 9. CMP AL,10111111B → 10111111B（0BFH）是X3数码管的位选码，**改此值可换扫描的最大位，核心！**：
;    - 改11011111B（0DFH）→ 扫描到X6数码管
;    - 改11101111B（0EFH）→ 扫描到X5数码管
; 10. JNZ AA1/MOV AL,11111110B → 位选码到X3后重置为X1，不可修改
; 11. LOOP AA0 → 循环CX次（显示3位数字），不可修改
; 12. JMP MAIN → 无限循环扫描，不可修改
MAIN:
	MOV AL,[POS]	            ; 读取当前位选码（不可改）
	MOV BX, 00H		        ; 段码表偏移量初始为0（改此值换起始段码）
	MOV CX,03H		        ; 扫描次数：3位数字（改此值换显示位数，核心！如2位→02H）
AA0:
	MOV SITE,AL	            ; 临时存储位选码到SITE（不可改）
	PUSH AX                  ; 保护AX（位选码）
	MOV AL,[BX+SI]           ; 读取段码表的第BX位段码（不可改）
	MOV NUM,AL               ; 存入NUM（不可改）
	POP AX                   ; 恢复AX（不可改）
 
	CALL DISP	            ; 调用显示子程序（不可改）
	INC BX                   ; 段码表偏移量+1（不可改）
	ROL AL,1	                ; 循环左移位选码（改ROR则扫描方向反转，核心！）	
	
	CMP AL,10111111B  	    ; 位选码到X3（0BFH）则重置（改此值换扫描最大位，核心！）
	JNZ AA1                  ; 未到则跳转到AA1（不可改）
	MOV AL,11111110B	        ; 重置位选码为X1（不可改）	
	
AA1:	
    LOOP AA0	                ; 循环CX次（不可改）
	JMP MAIN                 ; 无限循环扫描（不可改）

; ====================== 区块9：DISP显示子程序区【数码管驱动核心，硬件固定则无需改】 ======================
; 作用：将位选码输出到8255 A口、段码输出到8255 B口，调用DELAY子程序保证视觉暂留，是数码管能亮的关键
; 【可灵活修改的方式+效果（仅硬件接线调整，考试几乎不考）】：
; 1. MOV AL,[SITE]/MOV DX,A8255/OUT DX,AL → 输出位选码到A口，若位选接B口，需改DX=B8255
; 2. MOV AL,[NUM]/MOV DX,B8255/OUT DX,AL → 输出段码到B口，若段码接A口，需改DX=A8255
; 3. CALL DELAY → 调用短延时子程序，改DELAY的CX值可调整数码管亮度（见区块10）
DISP:
	PUSH AX                  ; 保护AX（不可改）
	PUSH DX                  ; 保护DX（不可改）
	MOV AL,[SITE]	        ; 读取临时存储的位选码（不可改）
    MOV DX,A8255             ; 指向8255 A口（位选）
    OUT DX,AL                ; 输出位选码（不可改）
    
    MOV AL,[NUM]	        ; 读取临时存储的段码（不可改）
    MOV DX,B8255             ; 指向8255 B口（段码）
    OUT DX,AL                ; 输出段码（不可改）
 
    CALL DELAY               ; 调用短延时，保证视觉暂留（改DELAY的CX值调亮度）
    POP DX                   ; 恢复DX（不可改）
    POP AX                   ; 恢复AX（不可改）
    RET  

; ====================== 区块10：短延时子程序区【数码管亮度调整，考试低频变种】 ======================
; 作用：软件短延时，让每个数码管的点亮状态保持一段时间，实现**人眼视觉暂留**（看起来3位同时亮），同时控制亮度
; 【每一行可灵活修改的方式+效果（亮度/刷新率调整，考试低频变种）】：
; 1. MOV CX,02FFH → 改此值调整延时长度，**核心！**：
;    - 改CX=01FFH → 更短延时 → 刷新率更高，亮度更低（无闪烁）
;    - 改CX=03FFH → 更长延时 → 刷新率更低，亮度更高（轻微闪烁）
;    - 改CX=0FFFFH → 极长延时 → 刷新率极低，明显闪烁，但亮度最高
DELAY:
    PUSH CX                  ; 保护CX（不可改）
    MOV CX,02FFH             ; 延时计数值（改此值调数码管亮度，核心！）
L1:
    LOOP L1                  ; 空循环延时（不可改）
    POP CX                   ; 恢复CX（不可改）
    RET

; ====================== 区块11：中断服务程序MIR6【位移控制核心，移动方向/范围变种改此部分！】 ======================

; 【每一行可灵活修改的方式+效果（移动方向/范围调整，考试高频变种！）】：
; 1. PUSH AX → 保护AX，不可修改
; 2. MOV AH, [POS] → 读取当前位选码到AH，不可修改
; 3. ROL AH,01H → 循环左移位选码（实现“123”右移），**改此指令可换移动方向，核心！**：
;    - 改ROR AH,01H → 循环右移位选码（实现“123”左移，考试高频变种！）
;    - 改SHL AH,1 → 逻辑左移（仅右移，无循环）
; 4. CMP AH,10111111B → 10111111B（0BFH）是X3数码管的位选码，**改此值可换位移的最大范围，核心！**：

MIR6:

    MOV AH, [POS]            ; 读取当前位选码（不可改）

	JNZ BB0                  ; 未到则跳转到BB0（不可改）
	MOV AH,11111110B         ; 重置位选码为X1（不可改）
	MOV POS, AH              ; 更新位选码（不可改）
	POP AX                   ; 恢复AX（不可改）
	IRET                     ; 中断返回（不可改）
BB0:
    MOV POS, AH              ; 更新位选码（不可改）
    POP AX                   ; 恢复AX（不可改）
    IRET                     ; 中断返回（不可改）
