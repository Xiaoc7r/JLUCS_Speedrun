; ====================== 区块1：硬件地址定义区【基础配置】 ======================
; 作用：给8255的A/B/C口和控制寄存器分配操作地址，是程序操作硬件的“门牌号”
; 【可灵活修改的行及效果】：
; 1. A8255 EQU 0640H → 修改0640H为其他十六进制数（如0600H/0200H），适配不同实验箱的8255 A口硬件地址
; 2. B8255 EQU 0642H → 修改0642H为其他值（如0602H/0202H），更换8255 B口的硬件地址（数据灯输出口）
; 3. C8255 EQU 0644H → 修改0644H为其他值（如0604H/0204H），更换8255 C口的硬件地址（预留接口）
; 4. MODE8255 EQU 0646H → 修改0646H为其他值（如0606H/0206H），更换8255控制寄存器的地址
A8255    EQU 0640H   ; 8255 A口地址：本程序未读取A口，预留作开关/按键输入
B8255    EQU 0642H   ; 8255 B口地址：输出控制数据灯亮灭的信号
C8255    EQU 0644H   ; 8255 C口地址：预留接口（可改作其他外设控制）
MODE8255 EQU 0646H   ; 8255控制寄存器地址：配置8255的工作模式

; ====================== 区块2：数据段定义区【数据存储】 ======================
; 作用：定义程序运行所需的全局变量，本区块仅定义标志位FLAG，用于控制移位方向
; 【可灵活修改的行及效果】：
; 1. FLAG DB 00H → 变量名可改（如DIR DB 00H，DIR代表方向），不影响功能仅提升可读性
; 2. 可新增变量：如添加DELAY_TIME DB 0FH，用于动态控制延时长度；添加DATA_BUF DB 80H，存储初始输出值
DATA SEGMENT
     FLAG DB 00H     ; 移位方向标志位：00H=默认/退出，01H=左移，02H=右移（可新增值扩展更多模式）
DATA ENDS

; ====================== 区块3：代码段声明区【汇编格式】 ======================
; 作用：汇编程序的固定格式，告诉编译器代码段、数据段的归属关系
; 【可灵活修改的行及效果】：无（汇编语法固定，修改会导致程序报错）
CODE SEGMENT
	ASSUME CS:CODE,DS:DATA  ; 绑定代码段寄存器CS到CODE段，数据段寄存器DS到DATA段

; ====================== 区块4：数据段初始化区【数据准备】 ======================
; 作用：将数据段的地址加载到DS寄存器，让程序能读写DATA段的变量（如FLAG）
; 【可灵活修改的行及效果】：无（数据段初始化的固定语法，仅需保证AX赋值为DATA段）
START:
    MOV AX,DATA     ; 将DATA段的段地址加载到AX寄存器
    MOV DS,AX       ; 将AX的值赋给DS，完成数据段的初始化

; ====================== 区块5：8255初始化区【硬件配置】 ======================
; 作用：配置8255的工作模式，指定A/B/C口的输入/输出属性，为本程序的灯控输出做准备
; 【可灵活修改的行及效果】：
; 1. MOV AL,90H → 修改90H为其他8255控制字，更换A/B口的输入输出模式：
;    - 90H：A口输入、B口输出、C口输出（本程序用此模式）
;    - 80H：A口输出、B口输出、C口输出（若A口也需控制外设可改）
;    - 98H：A口输入、B口输入、C口输出（若B口需读取数据可改）
;    - 88H：A口输出、B口输入、C口输入（适配其他外设输入需求）
    MOV DX,MODE8255 ; 将8255控制寄存器地址加载到DX
    MOV AL,90H      ; 8255工作模式控制字：A口输入，B口输出（核心配置）
    OUT DX,AL       ; 将控制字写入8255控制寄存器，完成硬件初始化

; ====================== 区块6：中断向量表设置区【中断入口配置】 ======================
; 作用：将MIR6、MIR7两个中断处理程序的地址写入CPU的中断向量表，让CPU能找到中断处理逻辑
; 【可灵活修改的行及效果】：
; 1. MOV SI,0038H → 修改0038H为其他中断向量地址，更换MIR6的中断号：
;    - 0038H对应中断号0EH（8259从片中断），0020H对应中断号08H（8259主片定时器中断）
;    - 0024H对应中断号09H（键盘中断），可根据实验需求更换中断触发源
; 2. MOV SI,003EH → 修改003EH为其他地址（如003CH/0040H），更换MIR7的中断向量存储位置，对应不同中断号
; 3. MOV AX,OFFSET MIR7 → 修改MIR7为其他子程序名（如MIR8），更换中断7的处理程序
; 4. MOV AX,OFFSET MIR6 → 修改MIR6为其他子程序名（如MIR5），更换中断6的处理程序
	PUSH DS          ; 保存当前DS寄存器的值，防止后续修改覆盖
	MOV AX,0000H     ; 将0000H加载到AX，指向中断向量表的基地址（0000H段）
	MOV DS,AX        ; 将DS设置为0000H，访问中断向量表
	
	; 设置MIR7中断的偏移地址和段地址到中断向量表
	MOV AX,OFFSET MIR7 ; 获取MIR7子程序的偏移地址
	MOV SI,003CH       ; 中断向量表中MIR7的偏移地址存储位置
	MOV [SI],AX        ; 将MIR7的偏移地址写入向量表
	MOV AX,CS          ; 获取代码段的段地址（MIR7在CS段）
	MOV SI,003EH       ; 中断向量表中MIR7的段地址存储位置
	MOV [SI],AX        ; 将MIR7的段地址写入向量表
	
	; 设置MIR6中断的偏移地址和段地址到中断向量表
	MOV AX,OFFSET MIR6 ; 获取MIR6子程序的偏移地址
	MOV SI,0038H       ; 中断向量表中MIR6的偏移地址存储位置
	MOV [SI],AX        ; 将MIR6的偏移地址写入向量表
	MOV AX,CS          ; 获取代码段的段地址（MIR6在CS段）
	MOV SI,003AH       ; 中断向量表中MIR6的段地址存储位置
	MOV [SI],AX        ; 将MIR6的段地址写入向量表
	POP DS           ; 恢复之前保存的DS寄存器值

; ====================== 区块7：8259中断控制器初始化区【中断使能】 ======================
; 作用：配置8259中断控制器的工作模式，开启指定中断的响应，让程序能接收中断信号
; 【可灵活修改的行及效果】：
; 1. MOV AL,11H → 修改11H为其他ICW1控制字：
;    - 11H：边沿触发、多片8259、需要ICW4
;    - 10H：边沿触发、单片8259、需要ICW4
;    - 01H：电平触发、单片8259、需要ICW4
; 2. MOV AL,08H → 修改08H为其他ICW2控制字，更换8259的中断号基址（如07H/09H）
; 3. MOV AL,04H → 修改04H为其他ICW3控制字，更换8259的级联方式（如02H/08H，适配不同主从片连接）
; 4. MOV AL,07H → 修改07H为其他ICW4控制字：
;    - 07H：非自动结束中断、缓冲模式、8086模式
;    - 03H：自动结束中断、缓冲模式、8086模式（简化中断处理）
; 5. MOV AL,2FH → 修改2FH为其他OCW1控制字，屏蔽/开启不同的中断源：
;    - 2FH：仅开启部分中断，00H=开启所有中断，FFH=屏蔽所有中断
    CLI             ; 关中断：防止初始化8259时被中断信号干扰
	
	MOV AL,11H      ; ICW1：8259初始化命令1，设置中断触发方式和级联模式
	OUT 20H,AL      ; 将ICW1写入8259主片的命令口（20H）
	MOV AL,08H      ; ICW2：8259初始化命令2，设置中断号基址
	OUT 21H,AL      ; 将ICW2写入8259主片的数据口（21H）
	MOV AL,04H      ; ICW3：8259初始化命令3，设置主从片级联方式
	OUT 21H,AL      ; 将ICW3写入8259主片数据口
	MOV AL,07H      ; ICW4：8259初始化命令4，设置中断结束方式和CPU模式
	OUT 21H,AL      ; 将ICW4写入8259主片数据口
	MOV AL,2FH      ; OCW1：8259操作命令1，设置中断屏蔽字（开启/屏蔽中断）
	OUT 21H,AL      ; 将OCW1写入8259主片数据口
	
	STI             ; 开中断：8259初始化完成，允许CPU响应中断

; ====================== 区块8：初始输出设置区【灯控初始状态】 ======================
; 作用：设置8255 B口的初始输出值，定义数据灯的初始亮灭状态
; 【可灵活修改的行及效果】：
; 1. MOV AL,80H → 修改80H为其他十六进制数，更换数据灯的初始亮灭状态：
;    - 80H：最高位灯亮，其余灯灭（1000 0000B）
;    - 01H：最低位灯亮，其余灯灭（0000 0001B）
;    - 0FH：低4位灯亮，高4位灯灭（0000 1111B）
;    - 0F0H：高4位灯亮，低4位灯灭（1111 0000B）
;    - 0FFH：所有灯全亮（1111 1111B）
;    - 00H：所有灯全灭（0000 0000B）
    MOV DX,B8255    ; 将8255 B口地址加载到DX
    MOV AL,80H      ; 初始输出值：最高位数据灯亮（可改此值定义初始灯态）
    OUT DX,AL       ; 将初始值写入8255 B口，设置数据灯初始状态

; ====================== 区块9：FLAG标志位初始化区【状态复位】 ======================
; 作用：将移位方向标志位FLAG初始化为00H，保证程序启动时的初始状态统一
; 【可灵活修改的行及效果】：
; 1. MOV FLAG,00H → 修改00H为其他值（如01H/02H），定义程序启动时的默认移位方向：
;    - 01H：启动后直接左移（无需MIR7中断触发）
;    - 02H：启动后直接右移（无需MIR6中断触发）
;    - 03H：可新增状态，如停止移位（需在主循环中添加对应判断逻辑）
    MOV FLAG,00H    ; 初始化FLAG为00H，程序启动时无移位方向指令

; ====================== 区块10：主循环核心功能区【移位控制】 ======================
; 作用：循环检测FLAG标志位的值，根据标志位控制数据灯的左移/右移，是程序的核心逻辑
; 【可灵活修改的行及效果】：
; 1. CMP FLAG,00H → 修改00H为其他值（如03H），更换退出/默认状态的标志位值
; 2. CMP FLAG,01H → 修改01H为其他值（如05H），更换左移状态的标志位值
; 3. ROR AL,1 → 修改ROR为其他移位指令，更换右移的实现方式：
;    - ROR AL,1：循环右移1位（每位依次右移，最低位移到最高位）
;    - SHR AL,1：逻辑右移1位（最高位补0，最低位移出）
;    - SAR AL,1：算术右移1位（最高位保持不变，最低位移出）
;    - ROR AL,2：循环右移2位（一次移两位，控制灯跳位亮灭）
; 4. ROL AL,1 → 修改ROL为其他移位指令，更换左移的实现方式：
;    - ROL AL,1：循环左移1位（每位依次左移，最高位移到最低位）
;    - SHL AL,1：逻辑左移1位（最低位补0，最高位移出）
;    - SAL AL,1：算术左移1位（同SHL，最低位补0）
;    - ROL AL,2：循环左移2位（一次移两位，跳位亮灭）
; 5. CALL DELAY → 可删除此行，取消延时（灯移位速度最大化）；也可调用其他延时子程序（如DELAY_LONG）
MAIN:
    CMP FLAG,00H    ; 判断FLAG是否为00H（默认/退出状态）
    JE EXIT         ; 若是，跳转到EXIT（直接输出当前值，不移位）
    CMP FLAG,01H    ; 判断FLAG是否为01H（左移状态，由MIR7中断触发）
    JE L            ; 若是，跳转到L（执行左移）
    ROR AL,1        ; FLAG为02H（右移状态），执行循环右移1位（核心移位指令）
    JMP EXIT        ; 跳转到EXIT，输出移位后的值
L:
    ROL AL,1        ; FLAG为01H，执行循环左移1位（核心移位指令）
    
EXIT:
    OUT DX,AL       ; 将移位/初始值写入8255 B口，更新数据灯状态
    CALL DELAY      ; 调用延时子程序，控制移位速度（灯亮灭的间隔）
    JMP MAIN        ; 无限循环，持续检测FLAG并控制移位

; ====================== 区块11：MIR6中断处理子程序区【右移触发】 ======================
; 作用：当MIR6中断触发时，将FLAG设置为02H，让主循环执行数据灯右移操作
; 【可灵活修改的行及效果】：
; 1. MOV FLAG,02H → 修改02H为其他值（如04H），更换右移状态的标志位值（需同步修改主循环的判断值）
; 2. 可新增代码：如MOV AL,00H → 在设置FLAG后清空输出值，中断触发时先灭灯再右移
MIR6:
    STI             ; 开中断：允许嵌套中断（若需响应其他中断）
    MOV FLAG,02H    ; 将FLAG设为02H，主循环检测后执行右移
	IRET            ; 中断返回：回到主循环继续执行

; ====================== 区块12：MIR7中断处理子程序区【左移触发】 ======================
; 作用：当MIR7中断触发时，将FLAG设置为01H，让主循环执行数据灯左移操作
; 【可灵活修改的行及效果】：
; 1. MOV FLAG,01H → 修改01H为其他值（如05H），更换左移状态的标志位值（需同步修改主循环的判断值）
; 2. 可新增代码：如OUT DX,AL → 在设置FLAG后立即更新灯态，加速左移响应
MIR7:
    STI             ; 开中断：允许嵌套中断
    MOV FLAG,01H    ; 将FLAG设为01H，主循环检测后执行左移
    IRET            ; 中断返回：回到主循环继续执行

; ====================== 区块13：延时子程序区【速度控制】 ======================
; 作用：通过循环消耗CPU时钟周期实现延时，控制数据灯移位的速度（延时越长，移位越慢）
; 【可灵活修改的行及效果】：
; 1. MOV CX,0F000H → 修改0F000H为其他十六进制数，调整延时长度：
;    - 0100H：短延时（移位速度极快，灯看起来全亮）
;    - 05000H：中等延时（移位速度适中，肉眼清晰可见）
;    - 0FFFFH：最长延时（移位速度极慢，适合观察单灯移动）
; 2. PUSH AX/POP AX → 可删除这两行，减少延时（循环内仅执行LOOP，耗时更短）；也可新增空操作（NOP）增加延时
; 3. 可新增外层循环：如添加MOV BX,05H → 在外层加BX循环，实现多级延时
DELAY:  
    PUSH CX         ; 保存CX寄存器的值，防止主循环的CX被覆盖
	MOV CX,0F000H   ; 延时计数值：值越大，延时越长（核心延时配置）
A0:		
    PUSH AX         ; 空操作：占用CPU周期，增加延时（可删）
	POP  AX         ; 空操作：占用CPU周期，增加延时（可删）
	LOOP A0         ; CX减1，直到CX=0时退出循环，完成延时
	POP CX	        ; 恢复CX寄存器的值
	RET             ; 子程序返回：回到主循环
